from pathlib import Path
import hashlib

import numpy as np
import pandas as pd


PHASE_POINTS = 121
COMPLETED_CYCLES = tuple(range(1, 25))


class CyclePreparationError(ValueError):
    """Raised when shared source data cannot be safely interpreted."""


def sha256_file(path):
    digest = hashlib.sha256()
    with Path(path).open("rb") as handle:
        for block in iter(lambda: handle.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def month_diff(start, end):
    return (end.year - start.year) * 12 + end.month - start.month


def prepare_cycle_curves(monthly_path, cycle_path, phase_points=PHASE_POINTS):
    """
    Prepare completed solar-cycle curves using one shared V2 convention.

    Returns:
        curves: {cycle_number: 1D numpy array}
        durations: {cycle_number: float}
        raw_peaks: {cycle_number: float}
        raw_peak_indices: {cycle_number: int}
        source_errors: {cycle_number: reason}
        phase_grid: 1D numpy array
        monthly_sha256, cycle_sha256

    Invalid individual cycles are recorded in source_errors.
    Invalid global source structure raises CyclePreparationError.
    No model fitting or experiment output is performed.
    """
    if not isinstance(phase_points, int) or phase_points < 2:
        raise CyclePreparationError("phase_points must be an integer >= 2.")

    monthly_path = Path(monthly_path)
    cycle_path = Path(cycle_path)

    if not monthly_path.is_file():
        raise CyclePreparationError(f"Monthly dataset not found: {monthly_path}")
    if not cycle_path.is_file():
        raise CyclePreparationError(f"Cycle summary not found: {cycle_path}")

    monthly_hash = sha256_file(monthly_path)
    cycle_hash = sha256_file(cycle_path)

    monthly = pd.read_csv(monthly_path)
    cycles = pd.read_csv(cycle_path)

    required_monthly = {"month_start", "ssn"}
    required_cycles = {
        "cycle_number",
        "status",
        "published_minimum_month",
        "months_expected_min_to_min",
    }

    missing_monthly = required_monthly - set(monthly.columns)
    missing_cycles = required_cycles - set(cycles.columns)

    if missing_monthly:
        raise CyclePreparationError(
            f"Missing monthly columns: {sorted(missing_monthly)}"
        )
    if missing_cycles:
        raise CyclePreparationError(
            f"Missing cycle columns: {sorted(missing_cycles)}"
        )

    monthly["month_start"] = pd.to_datetime(
        monthly["month_start"], errors="coerce"
    )
    monthly["ssn"] = pd.to_numeric(monthly["ssn"], errors="coerce")

    if monthly["month_start"].isna().any():
        raise CyclePreparationError("Monthly dataset contains invalid dates.")

    if not monthly["month_start"].is_unique:
        raise CyclePreparationError("Monthly dataset contains duplicate dates.")

    if not monthly["month_start"].eq(
        monthly["month_start"].dt.to_period("M").dt.to_timestamp()
    ).all():
        raise CyclePreparationError(
            "Monthly dates must be normalized to month-start timestamps."
        )

    monthly = monthly.sort_values("month_start").reset_index(drop=True)

    cycles["cycle_number"] = pd.to_numeric(
        cycles["cycle_number"], errors="coerce"
    )
    cycles["published_minimum_month"] = pd.to_datetime(
        cycles["published_minimum_month"], errors="coerce"
    )
    cycles["months_expected_min_to_min"] = pd.to_numeric(
        cycles["months_expected_min_to_min"], errors="coerce"
    )

    if cycles["cycle_number"].isna().any():
        raise CyclePreparationError("Cycle summary contains invalid cycle numbers.")

    cycle_numbers = cycles["cycle_number"].to_numpy(dtype=float)

    if not np.isfinite(cycle_numbers).all():
        raise CyclePreparationError(
            "Cycle numbers must all be finite."
        )

    if not np.equal(cycle_numbers, np.floor(cycle_numbers)).all():
        raise CyclePreparationError(
            "Cycle numbers must all be integers."
        )

    if cycles["cycle_number"].duplicated().any():
        raise CyclePreparationError("Cycle summary contains duplicate cycle numbers.")

    cycle_rows = {
        int(row["cycle_number"]): row
        for _, row in cycles.iterrows()
    }

    if 25 not in cycle_rows:
        raise CyclePreparationError("Cycle 25 row is required for Cycle 24 boundary.")

    cycle25 = cycle_rows[25]
    next_minimum = cycle25["published_minimum_month"]

    if pd.isna(next_minimum):
        raise CyclePreparationError(
            "Cycle 25 published minimum is required as Cycle 24 boundary."
        )

    starts = {}
    source_errors = {}

    for cycle in COMPLETED_CYCLES:
        row = cycle_rows.get(cycle)

        if row is None:
            source_errors[cycle] = "missing_cycle_summary_row"
            continue

        # Preserve a valid published minimum as a boundary independently
        # of whether this cycle is eligible as a training/heldout curve.
        if pd.isna(row["published_minimum_month"]):
            source_errors[cycle] = "missing_published_minimum"
            continue

        starts[cycle] = row["published_minimum_month"]

        if str(row["status"]).strip().lower() != "completed":
            source_errors[cycle] = "cycle_not_marked_completed"
            continue

        if pd.isna(row["months_expected_min_to_min"]):
            source_errors[cycle] = "missing_expected_duration"
            continue

        duration = float(row["months_expected_min_to_min"])
        if not np.isfinite(duration) or not duration.is_integer() or duration < 2:
            source_errors[cycle] = "invalid_expected_duration"
            continue

    starts[25] = next_minimum

    date_index = monthly.set_index("month_start")
    phase_grid = np.linspace(0.0, 1.0, phase_points)

    curves = {}
    durations = {}
    raw_peaks = {}
    raw_peak_indices = {}
    raw_peak_months = {}
    official_smoothed_max_strengths = {}
    official_smoothed_max_months = {}

    for cycle in COMPLETED_CYCLES:
        if cycle in source_errors:
            continue

        if cycle not in starts or (cycle + 1) not in starts:
            source_errors[cycle] = "missing_cycle_boundary"
            continue

        start = starts[cycle]
        end = starts[cycle + 1]
        row = cycle_rows[cycle]
        expected_duration = int(row["months_expected_min_to_min"])

        if pd.isna(start) or pd.isna(end) or end <= start:
            source_errors[cycle] = "invalid_cycle_boundaries"
            continue

        if month_diff(start, end) != expected_duration:
            source_errors[cycle] = "boundary_duration_mismatch"
            continue

        expected_months = pd.date_range(
            start, end, freq="MS", inclusive="left"
        )

        segment = monthly.loc[
            (monthly["month_start"] >= start)
            & (monthly["month_start"] < end)
        ].sort_values("month_start")

        if segment["month_start"].tolist() != list(expected_months):
            source_errors[cycle] = "missing_or_extra_monthly_rows"
            continue

        if len(segment) != expected_duration:
            source_errors[cycle] = "observed_duration_mismatch"
            continue

        values = segment["ssn"].to_numpy(dtype=float)

        if not np.isfinite(values).all():
            source_errors[cycle] = "non_finite_monthly_ssn"
            continue

        source_phase = np.linspace(0.0, 1.0, len(values))
        curves[cycle] = np.interp(phase_grid, source_phase, values)

        peak_index = int(np.argmax(values))
        durations[cycle] = float(expected_duration)
        raw_peaks[cycle] = float(values[peak_index])
        raw_peak_indices[cycle] = peak_index
        raw_peak_months[cycle] = (
            pd.Timestamp(segment.iloc[peak_index]["month_start"]).strftime("%Y-%m")
        )

        official_strength = row.get("official_smoothed_max_sn")
        official_month = row.get("official_smoothed_max_month")

        official_smoothed_max_strengths[cycle] = (
            None if pd.isna(official_strength) else float(official_strength)
        )
        official_smoothed_max_months[cycle] = (
            None
            if pd.isna(official_month) or not str(official_month).strip()
            else str(official_month)[:7]
        )

    return {
        "curves": curves,
        "durations": durations,
        "raw_peaks": raw_peaks,
        "raw_peak_indices": raw_peak_indices,
        "raw_peak_months": raw_peak_months,
        "official_smoothed_max_strengths": official_smoothed_max_strengths,
        "official_smoothed_max_months": official_smoothed_max_months,
        "source_errors": source_errors,
        "phase_grid": phase_grid,
        "phase_points": phase_points,
        "monthly_sha256": monthly_hash,
        "cycle_sha256": cycle_hash,
        "cycle_boundary_convention": (
            "Published minimum inclusive; next published minimum exclusive. "
            "Cycle 25 minimum is used only as Cycle 24 end boundary."
        ),
        "phase_convention": (
            "First observed month maps to phase 0; last observed month "
            "before the next minimum maps to phase 1; linear interpolation."
        ),
    }

from dataclasses import dataclass
from typing import Callable, Sequence
import numpy as np

from src.v3.solarshield_v3_kalman import ResidualStateKalmanFilter
from src.v3.solarshield_v3_enkf import ResidualStateEnsembleKalmanFilter


@dataclass(frozen=True)
class CycleFilterResult:
    cycle_id: int
    method: str
    residuals: tuple
    initialized_at: int | None
    snapshots: tuple
    missing_months: tuple
    seed: int | None = None
    ensemble_snapshots: tuple = ()


def _validate_covariance(value, shape, name):
    array = np.asarray(value, dtype=float)
    if array.shape != shape or not np.all(np.isfinite(array)):
        raise ValueError(f"Invalid {name} dimensions or values.")
    if not np.allclose(array, array.T, atol=1e-10):
        raise ValueError(f"{name} must be symmetric.")
    if np.linalg.eigvalsh(array).min() < -1e-9:
        raise ValueError(f"{name} must be positive semidefinite.")
    return array


def _has_valid_warmup(history, warmup_months):
    """Require consecutive finite observations before first residual initialization."""
    if len(history) < warmup_months:
        return False
    for value in history[-warmup_months:]:
        if value is None:
            return False
        try:
            numeric = float(value)
        except (TypeError, ValueError):
            return False
        if not np.isfinite(numeric):
            return False
    return True


def run_kf_cycle(
    cycle_id: int,
    observations: Sequence,
    baseline_predictor: Callable,
    initial_state_factory: Callable,
    process_covariance,
    observation_covariance,
    initial_covariance=((1.0, 0.0), (0.0, 1.0 / 36.0)),
    warmup_months: int = 6,
):
    """Run one independent cycle. Predictor receives past observations only."""
    if warmup_months != 6:
        raise ValueError("R19 requires exactly six warm-up months.")

    q = _validate_covariance(process_covariance, (2, 2), "Q")
    r = _validate_covariance(observation_covariance, (1, 1), "R")
    p0 = _validate_covariance(initial_covariance, (2, 2), "P0")

    history = []
    residual_records = []
    snapshots = []
    missing = []
    filter_instance = None
    initialized_at = None

    for index in range(len(observations)):
        # Warm-up observations are collected without residual assimilation.
        if index < warmup_months:
            history.append(observations[index])
            continue

        # The forecast is produced before the target observation is accessed.
        baseline = baseline_predictor(cycle_id, index, tuple(history))
        observed = observations[index]
        history.append(observed)

        if baseline is None or observed is None:
            missing.append(index + 1)
            if filter_instance is not None:
                filter_instance.predict()
                snapshots.append((index + 1, filter_instance.snapshot()))
            continue

        baseline = float(baseline)
        observed = float(observed)
        if not np.isfinite(baseline) or not np.isfinite(observed):
            missing.append(index + 1)
            if filter_instance is not None:
                filter_instance.predict()
                snapshots.append((index + 1, filter_instance.snapshot()))
            continue

        if filter_instance is None and not _has_valid_warmup(
            history[:-1], warmup_months
        ):
            continue

        residual = observed - baseline
        residual_records.append((index + 1, residual))

        if filter_instance is None:
            # Explicit protocol/owner decision required; no implicit x0.
            initial_state = np.asarray(
                initial_state_factory(residual), dtype=float
            ).reshape(-1)
            if initial_state.shape != (2,) or not np.all(np.isfinite(initial_state)):
                raise ValueError("Initializer must return a finite two-element state.")
            filter_instance = ResidualStateKalmanFilter(
                process_covariance=q,
                observation_covariance=r,
                initial_state=initial_state,
                initial_covariance=p0,
            )
            initialized_at = index + 1
            snapshots.append((index + 1, filter_instance.snapshot()))
        else:
            filter_instance.assimilate([residual])
            snapshots.append((index + 1, filter_instance.snapshot()))

    return CycleFilterResult(
        cycle_id=cycle_id,
        method="KF",
        residuals=tuple(residual_records),
        initialized_at=initialized_at,
        snapshots=tuple(snapshots),
        missing_months=tuple(missing),
    )


def run_enkf_cycle(
    cycle_id: int,
    observations: Sequence,
    baseline_predictor: Callable,
    initial_state_factory: Callable,
    process_covariance,
    observation_covariance,
    seed: int,
    ensemble_size: int,
    initial_covariance=((1.0, 0.0), (0.0, 1.0 / 36.0)),
    warmup_months: int = 6,
):
    """Run one independent EnKF cycle with reproducible initialization/noise."""
    if warmup_months != 6:
        raise ValueError("R19 requires exactly six warm-up months.")
    if ensemble_size not in (100, 250, 500):
        raise ValueError("R22 ensemble size must be 100, 250, or 500.")

    q = _validate_covariance(process_covariance, (2, 2), "Q")
    r = _validate_covariance(observation_covariance, (1, 1), "R")
    p0 = _validate_covariance(
        initial_covariance, (2, 2), "P0"
    )

    history = []
    residual_records = []
    snapshots = []
    ensemble_snapshots = []
    missing = []
    filter_instance = None
    initialized_at = None

    for index in range(len(observations)):
        if index < warmup_months:
            history.append(observations[index])
            continue

        baseline = baseline_predictor(cycle_id, index, tuple(history))
        observed = observations[index]
        history.append(observed)

        if baseline is None or observed is None:
            missing.append(index + 1)
            if filter_instance is not None:
                filter_instance.predict()
                snapshots.append((index + 1, filter_instance.mean.copy()))
                ensemble_snapshots.append((index + 1, filter_instance.ensemble.copy()))
            continue

        baseline = float(baseline)
        observed = float(observed)
        if not np.isfinite(baseline) or not np.isfinite(observed):
            missing.append(index + 1)
            if filter_instance is not None:
                filter_instance.predict()
                snapshots.append((index + 1, filter_instance.mean.copy()))
                ensemble_snapshots.append((index + 1, filter_instance.ensemble.copy()))
            continue

        if filter_instance is None and not _has_valid_warmup(
            history[:-1], warmup_months
        ):
            continue

        residual = observed - baseline
        residual_records.append((index + 1, residual))

        if filter_instance is None:
            initial_state = np.asarray(
                initial_state_factory(residual), dtype=float
            ).reshape(-1)
            if initial_state.shape != (2,) or not np.all(np.isfinite(initial_state)):
                raise ValueError("Initializer must return a finite two-element state.")

            init_rng = np.random.default_rng(
                np.random.SeedSequence([int(seed), 0])
            )
            ensemble = init_rng.multivariate_normal(
                initial_state, p0, size=ensemble_size
            )
            filter_instance = ResidualStateEnsembleKalmanFilter(
                ensemble=ensemble,
                process_covariance=q,
                observation_covariance=r,
                seed=int(np.random.SeedSequence([int(seed), 1]).generate_state(1)[0]),
            )
            initialized_at = index + 1
            snapshots.append((index + 1, filter_instance.mean.copy()))
            ensemble_snapshots.append((index + 1, filter_instance.ensemble.copy()))
        else:
            filter_instance.assimilate([residual])
            snapshots.append((index + 1, filter_instance.mean.copy()))
            ensemble_snapshots.append((index + 1, filter_instance.ensemble.copy()))

    return CycleFilterResult(
        cycle_id=cycle_id,
        method="EnKF",
        residuals=tuple(residual_records),
        initialized_at=initialized_at,
        snapshots=tuple(snapshots),
        missing_months=tuple(missing),
        seed=int(seed),
        ensemble_snapshots=tuple(ensemble_snapshots),
    )


def type8_calibration_quantiles(errors, probabilities=(0.50, 0.80, 0.95)):
    """Return Type-8 empirical absolute-error quantiles; never claim coverage."""
    values = np.asarray(errors, dtype=float).reshape(-1)
    values = values[np.isfinite(values)]
    if values.size == 0 or np.any(values < 0):
        raise ValueError("Calibration requires finite nonnegative absolute errors.")

    quantiles = np.quantile(
        values, probabilities, method="median_unbiased"
    )
    return {
        "quantile_method": "Hyndman-Fan Type 8",
        "nominal_levels": tuple(probabilities),
        "absolute_error_quantiles": tuple(float(x) for x in quantiles),
        "sample_count": int(values.size),
        "status": "EVIDENCE_LIMITED_NOT_VALIDATED",
        "coverage_guarantee": False,
    }


def compare_identical_origins(candidate_records, reference_records):
    """Require exact origin/target alignment before computing paired differences."""
    def keyed(records):
        result = {}
        for row in records:
            key = (row["cycle_id"], row["origin"], row["target"])
            if key in result:
                raise ValueError(f"Duplicate forecast key: {key}")
            result[key] = row
        return result

    candidate = keyed(candidate_records)
    reference = keyed(reference_records)

    if candidate.keys() != reference.keys():
        raise ValueError("R27 comparison requires identical forecast origins and targets.")

    return tuple({
        "key": key,
        "candidate_prediction": candidate[key]["prediction"],
        "reference_prediction": reference[key]["prediction"],
        "difference": (
            float(candidate[key]["prediction"])
            - float(reference[key]["prediction"])
        ),
    } for key in sorted(candidate))

"""Open-loop solar forecast construction using baseline plus residual correction."""

import math

import numpy as np


def _finite_array(value, shape, name):
    array = np.asarray(value, dtype=float)

    if array.shape != shape or not np.all(np.isfinite(array)):
        raise ValueError(f"Invalid {name} dimensions or values.")

    return array


def _validate_covariance(value, shape, name):
    array = _finite_array(value, shape, name)

    if not np.allclose(array, array.T, atol=1e-10):
        raise ValueError(f"{name} must be symmetric.")

    if np.linalg.eigvalsh(array).min() < -1e-9:
        raise ValueError(f"{name} must be positive semidefinite.")

    return array


def _positive_integer(value, name):
    if isinstance(value, bool):
        raise ValueError(f"{name} must be a positive integer.")

    try:
        result = int(value)
    except (TypeError, ValueError) as exc:
        raise ValueError(f"{name} must be a positive integer.") from exc

    if result != value or result <= 0:
        raise ValueError(f"{name} must be a positive integer.")

    return result


def _validate_metadata(cycle_id, origin):
    cycle = _positive_integer(cycle_id, "cycle_id")

    if not isinstance(origin, str) or not origin.strip():
        raise ValueError("origin must be a non-empty string.")

    return cycle, origin


def _prepare_baseline_forecasts(baseline_forecasts):
    if not baseline_forecasts:
        raise ValueError("At least one baseline forecast is required.")

    prepared = {}
    targets = set()

    for index, source in enumerate(baseline_forecasts):
        if not isinstance(source, dict):
            raise ValueError(
                f"Baseline forecast {index} must be a mapping."
            )

        if "observation" in source:
            raise ValueError(
                "Future observations must not enter forecast construction."
            )

        required = ("target", "horizon_months", "baseline_prediction")
        missing = [key for key in required if key not in source]

        if missing:
            raise ValueError(
                f"Baseline forecast {index} missing fields: {missing}"
            )

        target = source["target"]

        if not isinstance(target, str) or not target.strip():
            raise ValueError("target must be a non-empty string.")

        horizon = _positive_integer(
            source["horizon_months"], "horizon_months"
        )

        baseline = source["baseline_prediction"]

        if isinstance(baseline, bool):
            raise ValueError("baseline_prediction must be finite.")

        try:
            baseline = float(baseline)
        except (TypeError, ValueError) as exc:
            raise ValueError(
                "baseline_prediction must be finite."
            ) from exc

        if not math.isfinite(baseline):
            raise ValueError("baseline_prediction must be finite.")

        if horizon in prepared:
            raise ValueError(f"Duplicate forecast horizon: {horizon}")

        if target in targets:
            raise ValueError(f"Duplicate forecast target: {target}")

        prepared[horizon] = {
            "target": target,
            "baseline_prediction": baseline,
        }
        targets.add(target)

    return prepared


def _make_record(
    method,
    cycle_id,
    origin,
    horizon,
    baseline,
    correction,
    correction_sd,
    target,
):
    prediction = baseline + correction

    if not all(
        math.isfinite(value)
        for value in (prediction, correction, correction_sd)
    ):
        raise ValueError("Forecast construction produced non-finite values.")

    return {
        "method": method,
        "cycle_id": cycle_id,
        "origin": origin,
        "target": target,
        "horizon_months": horizon,
        "baseline_prediction": float(baseline),
        "residual_correction": float(correction),
        "residual_correction_sd": float(correction_sd),
        "prediction": float(prediction),
        "forecast_construction": "BASELINE_PLUS_RESIDUAL_CORRECTION",
        "forecast_mode": "OPEN_LOOP",
    }


def build_kf_forecasts(
    cycle_id,
    origin,
    state_at_origin,
    covariance_at_origin,
    baseline_forecasts,
    transition=((1.0, 1.0), (0.0, 1.0)),
    process_covariance=((1e-3, 0.0), (0.0, 1e-4)),
    observation=(1.0, 0.0),
):
    """Build open-loop KF forecasts from a filtered origin state."""

    cycle, origin = _validate_metadata(cycle_id, origin)

    state = _finite_array(
        state_at_origin, (2,), "state_at_origin"
    ).copy()

    covariance = _validate_covariance(
        covariance_at_origin, (2, 2), "covariance_at_origin"
    ).copy()

    transition = _finite_array(
        transition, (2, 2), "transition"
    )

    process_covariance = _validate_covariance(
        process_covariance, (2, 2), "process_covariance"
    )

    observation = _finite_array(
        observation, (2,), "observation"
    )

    if not np.any(observation):
        raise ValueError("Observation vector must not be all zero.")

    baselines = _prepare_baseline_forecasts(baseline_forecasts)
    records = []

    maximum_horizon = max(baselines)

    for horizon in range(1, maximum_horizon + 1):
        with np.errstate(over="ignore", invalid="ignore"):
            state = transition @ state
            covariance = (
                transition @ covariance @ transition.T
                + process_covariance
            )

        covariance = 0.5 * (covariance + covariance.T)

        if (
            not np.all(np.isfinite(state))
            or not np.all(np.isfinite(covariance))
        ):
            raise ValueError("KF forecast propagation became non-finite.")

        if horizon not in baselines:
            continue

        correction = float(observation @ state)
        correction_variance = float(
            observation @ covariance @ observation.T
        )

        if correction_variance < -1e-9:
            raise ValueError("KF correction variance is negative.")

        correction_sd = math.sqrt(max(0.0, correction_variance))
        baseline = baselines[horizon]

        records.append(
            _make_record(
                "KF",
                cycle,
                origin,
                horizon,
                baseline["baseline_prediction"],
                correction,
                correction_sd,
                baseline["target"],
            )
        )

    return tuple(records)


def build_enkf_forecasts(
    cycle_id,
    origin,
    ensemble_at_origin,
    baseline_forecasts,
    seed,
    transition=((1.0, 1.0), (0.0, 1.0)),
    process_covariance=((1e-3, 0.0), (0.0, 1e-4)),
    observation=(1.0, 0.0),
):
    """Build open-loop EnKF forecasts from an origin ensemble."""

    cycle, origin = _validate_metadata(cycle_id, origin)

    ensemble = np.asarray(ensemble_at_origin, dtype=float).copy()

    if (
        ensemble.ndim != 2
        or ensemble.shape[0] < 2
        or ensemble.shape[1] != 2
        or not np.all(np.isfinite(ensemble))
    ):
        raise ValueError(
            "ensemble_at_origin must be finite and contain at least 2 ensemble members."
        )

    transition = _finite_array(
        transition, (2, 2), "transition"
    )

    process_covariance = _validate_covariance(
        process_covariance, (2, 2), "process_covariance"
    )

    observation = _finite_array(
        observation, (2,), "observation"
    )

    if not np.any(observation):
        raise ValueError("Observation vector must not be all zero.")

    if isinstance(seed, (bool, np.bool_)) or not isinstance(
        seed, (int, np.integer)
    ):
        raise ValueError("seed must be a nonnegative integer.")

    seed = int(seed)

    if seed < 0:
        raise ValueError("seed must be a nonnegative integer.")

    baselines = _prepare_baseline_forecasts(baseline_forecasts)
    rng = np.random.default_rng(seed)
    records = []

    maximum_horizon = max(baselines)

    for horizon in range(1, maximum_horizon + 1):
        noise = rng.multivariate_normal(
            np.zeros(2),
            process_covariance,
            size=ensemble.shape[0],
        )

        with np.errstate(over="ignore", invalid="ignore"):
            ensemble = ensemble @ transition.T + noise

        if not np.all(np.isfinite(ensemble)):
            raise ValueError("EnKF forecast propagation became non-finite.")

        if horizon not in baselines:
            continue

        corrections = ensemble @ observation
        correction = float(np.mean(corrections))
        correction_sd = float(np.std(corrections, ddof=1))

        baseline = baselines[horizon]

        records.append(
            _make_record(
                "EnKF",
                cycle,
                origin,
                horizon,
                baseline["baseline_prediction"],
                correction,
                correction_sd,
                baseline["target"],
            )
        )

    return tuple(records)

def _positive_integer(value, name):
    if isinstance(value, bool) or not isinstance(value, (int, np.integer)):
        raise ValueError(f"{name} must be an integer.")
    if int(value) <= 0:
        raise ValueError(f"{name} must be positive.")
    return int(value)


def _month_number(label, name):
    if not isinstance(label, str):
        raise ValueError(f"{name} must use YYYY-MM format.")
    try:
        parsed = date.fromisoformat(label + "-01")
    except ValueError as exc:
        raise ValueError(f"{name} must use YYYY-MM format.") from exc
    if parsed.strftime("%Y-%m") != label:
        raise ValueError(f"{name} must use YYYY-MM format.")
    return parsed.year * 12 + parsed.month


def _validate_baselines(origin_label, origin_month, baseline_forecasts):
    if not isinstance(baseline_forecasts, (list, tuple)) or not baseline_forecasts:
        raise ValueError("baseline_forecasts must be a non-empty list or tuple.")

    origin_absolute = _month_number(origin_label, "origin_label")
    prepared = []
    seen_horizons = set()
    seen_targets = set()

    for index, row in enumerate(baseline_forecasts):
        if not isinstance(row, dict):
            raise ValueError(f"Baseline row {index} must be a mapping.")

        if "observation" in row:
            raise ValueError("Future observations must not enter forecast construction.")

        if not all(key in row for key in (
            "target", "horizon_months", "baseline_prediction"
        )):
            raise ValueError(f"Baseline row {index} is missing required fields.")

        horizon = _positive_integer(row["horizon_months"], "horizon_months")
        target = row["target"]

        if horizon in seen_horizons:
            raise ValueError(f"Duplicate forecast horizon: {horizon}")
        if target in seen_targets:
            raise ValueError(f"Duplicate forecast target: {target}")

        target_absolute = _month_number(target, "target")
        if target_absolute - origin_absolute != horizon:
            raise ValueError(
                f"Target {target} does not match origin {origin_label} "
                f"plus horizon {horizon}."
            )

        seen_horizons.add(horizon)
        seen_targets.add(target)
        prepared.append(dict(row))

    return prepared


def _origin_snapshot(result, method, origin_month):
    if getattr(result, "method", None) != method:
        raise ValueError(f"Expected a {method} CycleFilterResult.")

    month = _positive_integer(origin_month, "origin_month")
    collection_name = "snapshots" if method == "KF" else "ensemble_snapshots"
    collection = getattr(result, collection_name, None)

    if collection is None:
        raise ValueError(f"Result has no {collection_name} collection.")

    matches = []
    seen = set()

    for item in collection:
        if not isinstance(item, (tuple, list)) or len(item) != 2:
            raise ValueError(f"Malformed {collection_name} entry.")

        key, value = item
        key = _positive_integer(key, "snapshot month")

        if key in seen:
            raise ValueError(f"Duplicate {method} snapshot month: {key}")
        seen.add(key)

        if key == month:
            matches.append(value)

    if len(matches) != 1:
        raise ValueError(
            f"Expected exactly one {method} snapshot at month {month}; "
            f"found {len(matches)}."
        )

    return matches[0]

"""R21 shared physical Kalman noise parameter estimation."""

from dataclasses import dataclass
from itertools import product

import numpy as np
import scipy
from scipy.optimize import minimize


Q_SHAPE = np.array([[1.0, 0.5], [0.5, 1.0]])
R_SHAPE = np.array([[1.0]])
P0 = np.diag([1.0, 1.0 / 36.0])
F = np.array([[1.0, 1.0], [0.0, 1.0]])
H = np.array([[1.0, 0.0]])

LOG_BOUNDS = (-13.8155, 4.6052)
STARTS = (0.0001, 0.01, 1.0)
SENSITIVITY_LOG_TOLERANCE = 0.05


class R21EstimationError(RuntimeError):
    pass


@dataclass
class Sequence:
    values: np.ndarray
    scale: float
    valid_count: int


def robust_scale(values):
    finite = values[np.isfinite(values)]
    if finite.size == 0:
        raise R21EstimationError("UNRESOLVED_RESIDUAL_SCALE")

    median = np.median(finite)
    mad = np.median(np.abs(finite - median))
    scale = 1.4826 * mad

    if not np.isfinite(scale) or scale <= 0:
        q25, q75 = np.percentile(finite, [25, 75])
        scale = (q75 - q25) / 1.349

    if not np.isfinite(scale) or scale <= 0:
        raise R21EstimationError("UNRESOLVED_RESIDUAL_SCALE")

    return float(scale)


def prepare_sequences(residual_sequences):
    prepared = []

    for raw in residual_sequences:
        values = np.asarray(list(raw), dtype=float)

        if values.ndim != 1:
            raise R21EstimationError("Sequences must be one-dimensional.")

        valid = int(np.isfinite(values).sum())
        if valid < 12:
            raise R21EstimationError(
                "Minimum 12 valid observations required."
            )

        scale = robust_scale(values)
        prepared.append(Sequence(values / scale, scale, valid))

    if len(prepared) < 3:
        raise R21EstimationError(
            "Minimum three eligible sequences required."
        )

    return prepared


def sequence_nll(normalized_values, scale, q_physical, r_physical):
    values = np.asarray(normalized_values, dtype=float)

    if values.ndim != 1:
        raise R21EstimationError("Sequence must be one-dimensional.")

    if (
        not np.isfinite(q_physical)
        or not np.isfinite(r_physical)
        or q_physical <= 0
        or r_physical <= 0
        or not np.isfinite(scale)
        or scale <= 0
    ):
        raise R21EstimationError("Invalid physical q/r or scale.")

    scale_squared = scale * scale
    Q = (q_physical / scale_squared) * Q_SHAPE
    R = np.array([[r_physical / scale_squared]])

    x = None
    covariance = P0.copy()
    normalized_nll = 0.0
    scored_count = 0

    for observation in values:
        if x is None:
            if not np.isfinite(observation):
                continue

            # First finite residual initializes the state only.
            x = np.array([[float(observation)], [0.0]])
            continue

        # Preserve every subsequent calendar transition,
        # including months with missing observations.
        x = F @ x
        covariance = F @ covariance @ F.T + Q
        covariance = 0.5 * (covariance + covariance.T)

        if not np.isfinite(observation):
            continue

        innovation = float(observation - (H @ x)[0, 0])
        innovation_variance = float(
            (H @ covariance @ H.T + R)[0, 0]
        )

        if (
            not np.isfinite(innovation_variance)
            or innovation_variance <= 0
        ):
            raise R21EstimationError("Invalid innovation variance.")

        gain = covariance @ H.T / innovation_variance
        x = x + gain * innovation

        identity = np.eye(2)
        residual_operator = identity - gain @ H
        covariance = (
            residual_operator @ covariance @ residual_operator.T
            + gain @ R @ gain.T
        )
        covariance = 0.5 * (covariance + covariance.T)

        if (
            not np.all(np.isfinite(covariance))
            or np.linalg.eigvalsh(covariance).min() < -1e-9
        ):
            raise R21EstimationError("Invalid covariance.")

        normalized_nll += 0.5 * (
            np.log(2.0 * np.pi * innovation_variance)
            + innovation ** 2 / innovation_variance
        )
        scored_count += 1

    if x is None:
        raise R21EstimationError("No finite initialization observation.")

    if not np.isfinite(normalized_nll):
        raise R21EstimationError("Non-finite likelihood.")

    # Only observations after initialization contribute to physical NLL.
    physical_nll = normalized_nll + scored_count * np.log(scale)

    if not np.isfinite(physical_nll):
        raise R21EstimationError("Non-finite physical likelihood.")

    return float(physical_nll)

def objective(log_parameters, sequences):
    try:
        q_physical, r_physical = np.exp(log_parameters)
        return sum(
            sequence_nll(
                item.values,
                item.scale,
                q_physical,
                r_physical,
            )
            for item in sequences
        )
    except (
        ValueError,
        FloatingPointError,
        OverflowError,
        np.linalg.LinAlgError,
        R21EstimationError,
    ):
        return 1e100


def numerical_hessian(function, point, step=1e-4):
    point = np.asarray(point, dtype=float)
    dimension = point.size
    hessian = np.zeros((dimension, dimension), dtype=float)
    center = function(point)

    if not np.isfinite(center) or center >= 1e99:
        return None

    for i in range(dimension):
        ei = np.zeros(dimension)
        ei[i] = step

        fp = function(point + ei)
        fm = function(point - ei)

        if not np.isfinite(fp) or not np.isfinite(fm):
            return None

        hessian[i, i] = (fp - 2.0 * center + fm) / (step ** 2)

        for j in range(i + 1, dimension):
            ej = np.zeros(dimension)
            ej[j] = step

            fpp = function(point + ei + ej)
            fpm = function(point + ei - ej)
            fmp = function(point - ei + ej)
            fmm = function(point - ei - ej)

            if not all(np.isfinite(v) for v in (fpp, fpm, fmp, fmm)):
                return None

            value = (fpp - fpm - fmp + fmm) / (4.0 * step ** 2)
            hessian[i, j] = value
            hessian[j, i] = value

    return hessian


def estimate_r21_noise(residual_sequences):
    sequences = prepare_sequences(residual_sequences)
    diagnostics = []

    for initial_q, initial_r in product(STARTS, repeat=2):
        result = minimize(
            objective,
            x0=np.log([initial_q, initial_r]),
            args=(sequences,),
            method="L-BFGS-B",
            bounds=[LOG_BOUNDS, LOG_BOUNDS],
            options={
                "maxiter": 10000,
                "ftol": 1e-9,
                "gtol": 1e-6,
            },
        )

        parameters = np.asarray(result.x, dtype=float)
        finite = (
            np.all(np.isfinite(parameters))
            and np.isfinite(result.fun)
            and result.fun < 1e99
        )
        valid = bool(result.success and finite)

        boundary = bool(
            np.any(np.abs(parameters - LOG_BOUNDS[0]) < 1e-4)
            or np.any(np.abs(parameters - LOG_BOUNDS[1]) < 1e-4)
        )

        gradient = getattr(result, "jac", None)
        gradient_norm = (
            float(np.linalg.norm(gradient))
            if gradient is not None and np.all(np.isfinite(gradient))
            else None
        )

        diagnostics.append({
            "initial_q": initial_q,
            "initial_r": initial_r,
            "success": bool(result.success),
            "valid": valid,
            "status": int(result.status),
            "message": str(result.message),
            "objective": float(result.fun),
            "q_physical": (
                float(np.exp(parameters[0])) if finite else None
            ),
            "r_physical": (
                float(np.exp(parameters[1])) if finite else None
            ),
            "iterations": int(result.nit),
            "gradient_norm": gradient_norm,
            "boundary_hit": boundary,
        })

    successful = [item for item in diagnostics if item["valid"]]
    if not successful:
        raise R21EstimationError("ALL_OPTIMIZER_STARTS_FAILED")

    best = min(successful, key=lambda item: item["objective"])

    q_values = np.log([item["q_physical"] for item in successful])
    r_values = np.log([item["r_physical"] for item in successful])

    q_spread = float(np.ptp(q_values))
    r_spread = float(np.ptp(r_values))

    initialization_sensitive = bool(
        q_spread > SENSITIVITY_LOG_TOLERANCE
        or r_spread > SENSITIVITY_LOG_TOLERANCE
    )

    best_log_parameters = np.log([
        best["q_physical"],
        best["r_physical"],
    ])

    hessian = numerical_hessian(
        lambda p: objective(p, sequences),
        best_log_parameters,
    )

    curvature = {
        "status": "UNAVAILABLE",
        "eigenvalues": None,
        "condition_number": None,
    }

    if hessian is not None:
        eigenvalues = np.linalg.eigvalsh(hessian)
        condition_number = float(np.linalg.cond(hessian))

        curvature = {
            "status": (
                "POSITIVE_DEFINITE_LOCAL_CURVATURE"
                if np.all(eigenvalues > 0)
                else "NON_POSITIVE_LOCAL_CURVATURE"
            ),
            "eigenvalues": eigenvalues.tolist(),
            "condition_number": condition_number,
        }

    q_physical = best["q_physical"]
    r_physical = best["r_physical"]

    sequence_covariances = []
    for item in sequences:
        scale_squared = item.scale ** 2
        sequence_covariances.append({
            "scale": item.scale,
            "Q_normalized": (
                q_physical / scale_squared * Q_SHAPE
            ).tolist(),
            "R_normalized": (
                r_physical / scale_squared * R_SHAPE
            ).tolist(),
            "P0_normalized": P0.tolist(),
        })

    return {
        "protocol_reference": "R21_OPTION_A",
        "status": "ESTIMATION_COMPLETED_REQUIRES_REVIEW",
        "parameter_interpretation": "SHARED_PHYSICAL_Q_AND_R",
        "q_physical": q_physical,
        "r_physical": r_physical,
        "Q_physical": (q_physical * Q_SHAPE).tolist(),
        "R_physical": (r_physical * R_SHAPE).tolist(),
        "P0_normalized": P0.tolist(),
        "sequence_covariances": sequence_covariances,
        "residual_scales": [item.scale for item in sequences],
        "sequence_valid_counts": [
            item.valid_count for item in sequences
        ],
        "sequence_count": len(sequences),
        "total_valid_observations": sum(
            item.valid_count for item in sequences
        ),
        "objective": best["objective"],
        "optimizer": "L-BFGS-B",
        "scipy_version": scipy.__version__,
        "initialization_count": 9,
        "successful_initializations": len(successful),
        "boundary_hit": best["boundary_hit"],
        "boundary_hits_all_starts": sum(
            item["boundary_hit"] for item in diagnostics
        ),
        "initialization_sensitive": initialization_sensitive,
        "parameter_log_spread": {
            "q": q_spread,
            "r": r_spread,
        },
        "local_curvature": curvature,
        "identifiability_status": "NOT_ESTABLISHED",
        "start_diagnostics": diagnostics,
        "historical_data_used": False,
        "forecast_generated": False,
    }

"""Causal cycle-start detection using trailing-only observations."""
from collections import deque
from dataclasses import dataclass
import math
import numbers

import numpy as np


@dataclass(frozen=True)
class CycleDetectionEvent:
    detection_index: int
    smoothed_value: float
    confirmation_count: int
    sequence_id: int = 1


class CausalCycleStartDetector:
    """Causal detector with optional lock, decline, and re-arm behavior."""

    def __init__(
        self,
        smoothing_window=13,
        trend_lag=6,
        confirmation_months=6,
        minimum_history=24,
        reset_threshold=None,
        decline_confirmation_months=3,
        minimum_reset_duration=6,
    ):
        values = (
            smoothing_window,
            trend_lag,
            confirmation_months,
            minimum_history,
            decline_confirmation_months,
            minimum_reset_duration,
        )

        if any(
            isinstance(v, bool) or not isinstance(v, int) or v < 1
            for v in values
        ):
            raise ValueError("All integer settings must be positive integers.")

        if minimum_history < smoothing_window + trend_lag:
            raise ValueError(
                "minimum_history must cover smoothing_window + trend_lag."
            )

        if reset_threshold is not None:
            if (
                isinstance(reset_threshold, bool)
                or not isinstance(reset_threshold, (int, float))
                or not math.isfinite(reset_threshold)
                or reset_threshold <= 0
            ):
                raise ValueError("reset_threshold must be finite and positive.")

        self.smoothing_window = smoothing_window
        self.trend_lag = trend_lag
        self.confirmation_months = confirmation_months
        self.minimum_history = minimum_history
        self.reset_threshold = (
            None if reset_threshold is None else float(reset_threshold)
        )
        self.decline_confirmation_months = decline_confirmation_months
        self.minimum_reset_duration = minimum_reset_duration

        capacity = minimum_history + smoothing_window + trend_lag
        self.values = deque(maxlen=capacity)
        self.smoothed = deque(maxlen=capacity)

        self.index = -1
        self.positive_run = 0
        self.event = None
        self.event_sequence_id = 0

        self.state = "ARMED"
        self.running_peak = None
        self.decline_start_index = None
        self.decline_confirmation_count = 0
        self.reset_duration_count = 0
        self.missing_count = 0

    def _update_locked_state(self, smooth):
        if self.running_peak is None:
            self.running_peak = smooth

        self.running_peak = max(self.running_peak, smooth)

        decline = self.running_peak - smooth
        if decline >= self.reset_threshold:
            if self.state != "RESET_PENDING":
                self.state = "RESET_PENDING"
                self.decline_start_index = self.index
                self.decline_confirmation_count = 1
            else:
                self.decline_confirmation_count += 1

            self.reset_duration_count = (
                self.index - self.decline_start_index + 1
            )

            if (
                self.decline_confirmation_count
                >= self.decline_confirmation_months
                and self.reset_duration_count
                >= self.minimum_reset_duration
            ):
                self.state = "REARMED"
                self.positive_run = 0
        else:
            self.state = "LOCKED"
            self.decline_start_index = None
            self.decline_confirmation_count = 0
            self.reset_duration_count = 0

    def update(self, value):
        # Missing values occupy a calendar position but are not observations.
        if value is None:
            self.index += 1
            self.missing_count += 1
            self.values.append(None)
            self.smoothed.append(None)

            if self.state == "ARMED":
                self.positive_run = 0

            return None

        if (
            isinstance(value, (bool, np.bool_))
            or not isinstance(value, numbers.Real)
        ):
            raise ValueError("Input must be finite and non-negative.")

        value = float(value)

        if not math.isfinite(value) or value < 0:
            raise ValueError("Input must be finite and non-negative.")

        self.index += 1
        self.missing_count = 0
        self.values.append(value)

        if len(self.values) < self.smoothing_window:
            self.smoothed.append(None)
            return None

        recent = list(self.values)[-self.smoothing_window:]

        if any(item is None for item in recent):
            self.smoothed.append(None)
            if self.state == "ARMED":
                self.positive_run = 0
            return None

        smooth = sum(item / self.smoothing_window for item in recent)
        self.smoothed.append(smooth)

        if len(self.values) < self.minimum_history:
            return None

        if len(self.smoothed) <= self.trend_lag:
            return None

        history = list(self.smoothed)
        previous_smooth = history[-1 - self.trend_lag]

        # Trend comparison must use the exact calendar-month lag.
        if history[-1] is None or previous_smooth is None:
            if self.state == "ARMED":
                self.positive_run = 0
            return None

        rising = history[-1] > previous_smooth

        if self.state == "REARMED":
            self.state = "ARMED"
            self.positive_run = 0
            self.running_peak = None
            self.decline_start_index = None
            self.decline_confirmation_count = 0
            self.reset_duration_count = 0
            return None

        if self.state in ("LOCKED", "RESET_PENDING"):
            if self.reset_threshold is not None:
                self._update_locked_state(smooth)
            return None

        self.positive_run = self.positive_run + 1 if rising else 0

        if self.positive_run >= self.confirmation_months:
            self.event_sequence_id += 1
            self.event = CycleDetectionEvent(
                detection_index=self.index,
                smoothed_value=smooth,
                confirmation_count=self.positive_run,
                sequence_id=self.event_sequence_id,
            )
            self.state = "LOCKED"

"""Dynamic time warping with optional normalized phase constraints."""
import math
import numbers

import numpy as np


def dtw_distance(
    reference,
    candidate,
    window=None,
    return_path=False,
    phase_window=None,
    return_normalized=False,
):
    """Compute absolute-cost DTW distance.

    Parameters
    ----------
    reference, candidate:
        Finite, non-empty one-dimensional numeric sequences.
    window:
        Optional Sakoe-Chiba index window. The effective window is widened
        to at least the sequence-length difference for endpoint reachability.
    return_path:
        Return the optimal index-pair path.
    phase_window:
        Optional maximum normalized phase displacement in [0, 1].
        Both sequences are mapped linearly to phase [0, 1].
    return_normalized:
        Also return raw_cost / number_of_path_cells.

    Return compatibility
    --------------------
    Default: raw_cost
    return_path=True: (raw_cost, path)
    return_normalized=True: (raw_cost, normalized_cost)
    Both options: (raw_cost, normalized_cost, path)
    """
    a = np.asarray(reference, dtype=float)
    b = np.asarray(candidate, dtype=float)

    if a.ndim != 1 or b.ndim != 1 or a.size == 0 or b.size == 0:
        raise ValueError("DTW inputs must be non-empty one-dimensional sequences.")
    if not np.all(np.isfinite(a)) or not np.all(np.isfinite(b)):
        raise ValueError("DTW inputs must contain only finite values.")

    n = int(a.size)
    m = int(b.size)

    if not isinstance(return_path, (bool, np.bool_)):
        raise ValueError("return_path must be boolean.")
    if not isinstance(return_normalized, (bool, np.bool_)):
        raise ValueError("return_normalized must be boolean.")

    if window is None:
        index_window = max(n, m)
    else:
        if (
            isinstance(window, (bool, np.bool_))
            or not isinstance(window, numbers.Integral)
            or window < 0
        ):
            raise ValueError("window must be a non-negative integer or None.")
        index_window = max(int(window), abs(n - m))

    if phase_window is not None:
        if (
            isinstance(phase_window, (bool, np.bool_))
            or not isinstance(phase_window, numbers.Real)
            or not math.isfinite(phase_window)
            or not 0.0 <= phase_window <= 1.0
        ):
            raise ValueError("phase_window must be a finite number in [0, 1].")
        phase_window = float(phase_window)

    costs = np.full((n + 1, m + 1), np.inf, dtype=float)
    parents = np.full((n, m), -1, dtype=np.int8)
    costs[0, 0] = 0.0

    for i in range(n):
        j_start = max(0, i - index_window)
        j_end = min(m, i + index_window + 1)

        for j in range(j_start, j_end):
            if phase_window is not None:
                phase_a = i / (n - 1) if n > 1 else 0.0
                phase_b = j / (m - 1) if m > 1 else 0.0

                if abs(phase_a - phase_b) > phase_window + 1e-12:
                    continue

            diagonal = costs[i, j]
            vertical = costs[i, j + 1]
            horizontal = costs[i + 1, j]

            best_cost, direction = min(
                (diagonal, 0),
                (vertical, 1),
                (horizontal, 2),
                key=lambda item: (item[0], item[1]),
            )

            if math.isinf(best_cost):
                continue

            local_cost = abs(float(a[i]) - float(b[j]))
            total = best_cost + local_cost

            if not math.isfinite(total):
                raise FloatingPointError("DTW accumulated cost overflowed.")

            costs[i + 1, j + 1] = total
            parents[i, j] = direction

    raw_cost = float(costs[n, m])

    if not math.isfinite(raw_cost):
        raise ValueError("No valid DTW path satisfies the supplied constraints.")

    if not return_path and not return_normalized:
        return raw_cost

    path = []
    i = n - 1
    j = m - 1

    while True:
        path.append((i, j))

        if i == 0 and j == 0:
            break

        direction = int(parents[i, j])

        if direction == 0:
            i -= 1
            j -= 1
        elif direction == 1:
            i -= 1
        elif direction == 2:
            j -= 1
        else:
            raise RuntimeError("DTW path reconstruction failed.")

    path.reverse()
    normalized_cost = raw_cost / len(path)

    if return_normalized and return_path:
        return raw_cost, normalized_cost, path
    if return_normalized:
        return raw_cost, normalized_cost
    return raw_cost, path

"""Global and local peak detection for finite one-dimensional curves."""
from dataclasses import dataclass
import math
import numbers

import numpy as np


@dataclass(frozen=True)
class PeakPlateau:
    start_index: int
    end_index: int
    center_index: float
    center_phase: float
    amplitude: float
    width: int
    is_primary: bool


@dataclass(frozen=True)
class LocalPeak:
    start_index: int
    end_index: int
    center_index: float
    center_phase: float
    amplitude: float
    prominence: float
    width: int
    is_primary: bool


def _validate_curve(curve):
    values = np.asarray(curve, dtype=float)
    if values.ndim != 1 or values.size < 3:
        raise ValueError("Curve must be one-dimensional with at least three points.")
    if not np.all(np.isfinite(values)):
        raise ValueError("Curve values must be finite.")
    return values


def _validate_tolerance(value, name):
    if (
        isinstance(value, (bool, np.bool_))
        or not isinstance(value, numbers.Real)
        or not math.isfinite(value)
        or value < 0
    ):
        raise ValueError(f"{name} must be finite and non-negative.")
    return float(value)


def detect_peak_plateaus(curve, tolerance=0.0):
    """Backward-compatible regions near the global maximum."""
    values = np.asarray(curve, dtype=float)

    if values.ndim != 1 or values.size < 2:
        raise ValueError("Curve must be one-dimensional with at least two points.")
    if not np.all(np.isfinite(values)):
        raise ValueError("Curve values must be finite.")

    tolerance = _validate_tolerance(tolerance, "tolerance")
    maximum = float(np.max(values))
    selected = values >= maximum - tolerance

    regions = []
    start = None

    for index, included in enumerate(selected):
        if included and start is None:
            start = index
        elif not included and start is not None:
            regions.append((start, index - 1))
            start = None

    if start is not None:
        regions.append((start, values.size - 1))

    amplitudes = [float(np.max(values[a:b + 1])) for a, b in regions]
    primary = max(
        range(len(regions)),
        key=lambda i: (amplitudes[i], -regions[i][0]),
    )

    result = []
    denominator = values.size - 1

    for i, (a, b) in enumerate(regions):
        center = (a + b) / 2.0
        result.append(PeakPlateau(
            start_index=a,
            end_index=b,
            center_index=center,
            center_phase=center / denominator,
            amplitude=amplitudes[i],
            width=b - a + 1,
            is_primary=(i == primary),
        ))

    return tuple(result)


def detect_local_peaks(curve, prominence_threshold=0.0):
    """Detect interior local maxima and flat local-maximum plateaus.

    Prominence is measured relative to the higher of the left and right
    contour minima. Endpoints are not classified as local peaks.
    """
    values = _validate_curve(curve)
    threshold = _validate_tolerance(
        prominence_threshold, "prominence_threshold"
    )

    n = values.size
    regions = []
    i = 1

    while i < n - 1:
        start = i
        end = i

        while end + 1 < n and values[end + 1] == values[start]:
            end += 1

        if (
            values[start] > values[start - 1]
            and end < n - 1
            and values[end] > values[end + 1]
        ):
            regions.append((start, end, float(values[start])))

        i = end + 1

    detected = []

    for start, end, amplitude in regions:
        left_higher = [
            peak for peak in regions
            if peak[0] < start and peak[2] > amplitude
        ]
        right_higher = [
            peak for peak in regions
            if peak[0] > end and peak[2] > amplitude
        ]

        left_bound = (
            max(left_higher, key=lambda peak: peak[0])[1]
            if left_higher else 0
        )
        right_bound = (
            min(right_higher, key=lambda peak: peak[0])[0]
            if right_higher else n - 1
        )

        left_min = float(np.min(values[left_bound:start + 1]))
        right_min = float(np.min(values[end:right_bound + 1]))
        prominence = max(
            0.0,
            amplitude - max(left_min, right_min),
        )

        if prominence >= threshold:
            center = (start + end) / 2.0
            detected.append({
                "start": start,
                "end": end,
                "center": center,
                "amplitude": amplitude,
                "prominence": prominence,
            })

    if not detected:
        return tuple()

    primary_index = max(
        range(len(detected)),
        key=lambda k: (
            detected[k]["amplitude"],
            -detected[k]["start"],
        ),
    )

    result = []
    denominator = n - 1

    for k, peak in enumerate(detected):
        result.append(LocalPeak(
            start_index=peak["start"],
            end_index=peak["end"],
            center_index=peak["center"],
            center_phase=peak["center"] / denominator,
            amplitude=peak["amplitude"],
            prominence=peak["prominence"],
            width=peak["end"] - peak["start"] + 1,
            is_primary=(k == primary_index),
        ))

    return tuple(result)

"""Composite cycle-shape loss with legacy and local-peak diagnostics."""
from dataclasses import dataclass
import math
import numbers

import numpy as np

from .solarshield_v3_dtw import dtw_distance
from .solarshield_v3_peaks import (
    detect_peak_plateaus,
    detect_local_peaks,
)


@dataclass(frozen=True)
class CycleLossBreakdown:
    total: float
    shape_dtw: float
    peak_timing_phase_error: float
    peak_amplitude_absolute_error: float
    reference_peak_count: int = 1
    candidate_peak_count: int = 1
    multiple_peak_regions: bool = False
    shape_dtw_normalized: float = 0.0
    local_reference_peak_count: int = 0
    local_candidate_peak_count: int = 0
    matched_peak_count: int = 0
    unmatched_reference_peak_count: int = 0
    unmatched_candidate_peak_count: int = 0
    local_peak_timing_phase_error: float = 0.0
    local_peak_matching_cost: float = 0.0


def _symmetric_hausdorff_distance(reference_phases, candidate_phases):
    """Legacy symmetric Hausdorff distance between global peak phases."""
    a = np.asarray(reference_phases, dtype=float)
    b = np.asarray(candidate_phases, dtype=float)

    if a.size == 0 or b.size == 0:
        raise ValueError("Peak phase sets cannot be empty.")

    pairwise = np.abs(a[:, None] - b[None, :])
    directed_ab = float(np.max(np.min(pairwise, axis=1)))
    directed_ba = float(np.max(np.min(pairwise, axis=0)))
    return max(directed_ab, directed_ba)


def _validate_nonnegative_real(value, name):
    if (
        isinstance(value, (bool, np.bool_))
        or not isinstance(value, numbers.Real)
        or not math.isfinite(value)
        or value < 0
    ):
        raise ValueError(f"{name} must be finite and non-negative.")
    return float(value)


def _order_preserving_peak_match(
    reference_peaks,
    candidate_peaks,
    unmatched_penalty,
    max_phase_distance,
):
    """Dynamic-programming alignment of ordered peak phase sequences.

    Matching cost is absolute phase displacement. An unmatched peak incurs
    unmatched_penalty. Matches beyond max_phase_distance are prohibited.
    """
    a = [peak.center_phase for peak in reference_peaks]
    b = [peak.center_phase for peak in candidate_peaks]

    n = len(a)
    m = len(b)

    dp = np.full((n + 1, m + 1), np.inf, dtype=float)
    action = np.full((n + 1, m + 1), -1, dtype=np.int8)
    dp[0, 0] = 0.0

    for i in range(n + 1):
        for j in range(m + 1):
            current = dp[i, j]
            if not math.isfinite(current):
                continue

            if i < n and current + unmatched_penalty < dp[i + 1, j]:
                dp[i + 1, j] = current + unmatched_penalty
                action[i + 1, j] = 1

            if j < m and current + unmatched_penalty < dp[i, j + 1]:
                dp[i, j + 1] = current + unmatched_penalty
                action[i, j + 1] = 2

            if i < n and j < m:
                distance = abs(a[i] - b[j])
                allowed = (
                    max_phase_distance is None
                    or distance <= max_phase_distance + 1e-12
                )
                matched_cost = current + distance

                if allowed and matched_cost < dp[i + 1, j + 1]:
                    dp[i + 1, j + 1] = matched_cost
                    action[i + 1, j + 1] = 0

    matches = []
    i, j = n, m

    while i > 0 or j > 0:
        step = int(action[i, j])

        if step == 0:
            matches.append((i - 1, j - 1))
            i -= 1
            j -= 1
        elif step == 1:
            i -= 1
        elif step == 2:
            j -= 1
        else:
            raise RuntimeError("Peak matching path reconstruction failed.")

    matches.reverse()

    matched_reference = {i for i, _ in matches}
    matched_candidate = {j for _, j in matches}

    unmatched_reference = n - len(matched_reference)
    unmatched_candidate = m - len(matched_candidate)

    phase_errors = [
        abs(a[i] - b[j]) for i, j in matches
    ]

    mean_phase_error = (
        float(np.mean(phase_errors)) if phase_errors else 0.0
    )

    return {
        "matches": tuple(matches),
        "matched_count": len(matches),
        "unmatched_reference": unmatched_reference,
        "unmatched_candidate": unmatched_candidate,
        "mean_phase_error": mean_phase_error,
        "matching_cost": float(dp[n, m]),
    }


def composite_cycle_loss(
    reference,
    candidate,
    *,
    shape_weight=1.0,
    peak_timing_weight=1.0,
    peak_amplitude_weight=1.0,
    dtw_window=None,
    peak_tolerance=0.0,
    peak_prominence_fraction=0.05,
    unmatched_peak_penalty=0.1,
    max_peak_phase_distance=None,
    local_peak_weight=0.0,
    dtw_phase_window=None,
):
    y = np.asarray(reference, dtype=float).reshape(-1)
    p = np.asarray(candidate, dtype=float).reshape(-1)

    if y.size < 2 or p.size < 2:
        raise ValueError("Both curves need at least two points.")
    if not np.all(np.isfinite(y)) or not np.all(np.isfinite(p)):
        raise ValueError("Curves must contain finite values.")

    weights = (
        shape_weight,
        peak_timing_weight,
        peak_amplitude_weight,
        local_peak_weight,
    )

    validated_weights = [
        _validate_nonnegative_real(value, "loss weight")
        for value in weights
    ]
    (
        shape_weight,
        peak_timing_weight,
        peak_amplitude_weight,
        local_peak_weight,
    ) = validated_weights

    peak_tolerance = _validate_nonnegative_real(
        peak_tolerance, "peak_tolerance"
    )
    prominence_fraction = _validate_nonnegative_real(
        peak_prominence_fraction, "peak_prominence_fraction"
    )

    if prominence_fraction > 1.0:
        raise ValueError("peak_prominence_fraction must be in [0, 1].")

    unmatched_penalty = _validate_nonnegative_real(
        unmatched_peak_penalty, "unmatched_peak_penalty"
    )

    if max_peak_phase_distance is not None:
        max_peak_phase_distance = _validate_nonnegative_real(
            max_peak_phase_distance, "max_peak_phase_distance"
        )
        if max_peak_phase_distance > 1.0:
            raise ValueError("max_peak_phase_distance must be in [0, 1].")

    if dtw_phase_window is not None:
        dtw_phase_window = _validate_nonnegative_real(
            dtw_phase_window, "dtw_phase_window"
        )
        if dtw_phase_window > 1.0:
            raise ValueError("dtw_phase_window must be in [0, 1].")

    # Preserve the legacy global-maximum plateau diagnostics.
    reference_global = detect_peak_plateaus(
        y, tolerance=peak_tolerance
    )
    candidate_global = detect_peak_plateaus(
        p, tolerance=peak_tolerance
    )

    reference_global_phases = [
        peak.center_phase for peak in reference_global
    ]
    candidate_global_phases = [
        peak.center_phase for peak in candidate_global
    ]

    legacy_timing_error = _symmetric_hausdorff_distance(
        reference_global_phases,
        candidate_global_phases,
    )

    # Local peak prominence is relative to each curve's own dynamic range.
    reference_range = float(np.max(y) - np.min(y))
    candidate_range = float(np.max(p) - np.min(p))

    reference_local = detect_local_peaks(
        y,
        prominence_threshold=prominence_fraction * reference_range,
    )
    candidate_local = detect_local_peaks(
        p,
        prominence_threshold=prominence_fraction * candidate_range,
    )

    matching = _order_preserving_peak_match(
        reference_local,
        candidate_local,
        unmatched_penalty,
        max_peak_phase_distance,
    )

    raw_dtw, normalized_dtw = dtw_distance(
        y,
        p,
        window=dtw_window,
        phase_window=dtw_phase_window,
        return_normalized=True,
    )

    amplitude_error = abs(float(np.max(y)) - float(np.max(p)))

    multiple_global_peaks = (
        len(reference_global) > 1 or len(candidate_global) > 1
    )

    total = (
        shape_weight * raw_dtw
        + peak_timing_weight * legacy_timing_error
        + peak_amplitude_weight * amplitude_error
        + local_peak_weight * matching["matching_cost"]
    )

    if not math.isfinite(total):
        raise FloatingPointError("Composite cycle loss is non-finite.")

    return CycleLossBreakdown(
        total=float(total),
        shape_dtw=float(raw_dtw),
        peak_timing_phase_error=float(legacy_timing_error),
        peak_amplitude_absolute_error=float(amplitude_error),
        reference_peak_count=len(reference_global),
        candidate_peak_count=len(candidate_global),
        multiple_peak_regions=multiple_global_peaks,
        shape_dtw_normalized=float(normalized_dtw),
        local_reference_peak_count=len(reference_local),
        local_candidate_peak_count=len(candidate_local),
        matched_peak_count=matching["matched_count"],
        unmatched_reference_peak_count=matching["unmatched_reference"],
        unmatched_candidate_peak_count=matching["unmatched_candidate"],
        local_peak_timing_phase_error=matching["mean_phase_error"],
        local_peak_matching_cost=matching["matching_cost"],
    )

"""R26 calibration evidence diagnostics for V3 forecast records."""

from collections import defaultdict
import math

import numpy as np


_REQUIRED_FORECAST_FIELDS = (
    "method",
    "cycle_id",
    "origin",
    "target",
    "horizon_months",
    "prediction",
    "observation",
)

_REQUIRED_INTERVAL_FIELDS = (
    "method",
    "cycle_id",
    "origin",
    "target",
    "horizon_months",
    "nominal_level",
    "lower",
    "upper",
)

_NOMINAL_LEVELS = (0.50, 0.80, 0.95)


def _finite_number(value, field):
    if isinstance(value, bool):
        raise ValueError(f"{field} must be numeric and finite.")

    try:
        result = float(value)
    except (TypeError, ValueError) as exc:
        raise ValueError(f"{field} must be numeric and finite.") from exc

    if not math.isfinite(result):
        raise ValueError(f"{field} must be numeric and finite.")

    return result


def _positive_integer(value, field):
    if isinstance(value, bool):
        raise ValueError(f"{field} must be a positive integer.")

    try:
        number = int(value)
    except (TypeError, ValueError) as exc:
        raise ValueError(f"{field} must be a positive integer.") from exc

    if number != value or number <= 0:
        raise ValueError(f"{field} must be a positive integer.")

    return number


def _required_text(value, field):
    if not isinstance(value, str) or not value.strip():
        raise ValueError(f"{field} must be a non-empty string.")
    return value


def _forecast_key(row):
    return (
        row["method"],
        row["cycle_id"],
        row["origin"],
        row["target"],
        row["horizon_months"],
    )


def _validate_forecast_records(records):
    indexed = {}

    for index, source in enumerate(records):
        if not isinstance(source, dict):
            raise ValueError(f"Forecast record {index} must be a mapping.")

        missing = [
            field for field in _REQUIRED_FORECAST_FIELDS
            if field not in source
        ]
        if missing:
            raise ValueError(
                f"Forecast record {index} missing fields: {missing}"
            )

        row = {
            "method": _required_text(source["method"], "method"),
            "cycle_id": _positive_integer(source["cycle_id"], "cycle_id"),
            "origin": _required_text(source["origin"], "origin"),
            "target": _required_text(source["target"], "target"),
            "horizon_months": _positive_integer(
                source["horizon_months"], "horizon_months"
            ),
            "prediction": _finite_number(
                source["prediction"], "prediction"
            ),
            "observation": _finite_number(
                source["observation"], "observation"
            ),
        }

        key = _forecast_key(row)

        if key in indexed:
            raise ValueError(f"Duplicate forecast record key: {key}")

        row["absolute_error"] = abs(
            row["prediction"] - row["observation"]
        )
        indexed[key] = row

    if not indexed:
        raise ValueError("At least one forecast record is required.")

    return indexed


def _validate_interval_records(interval_records, forecasts):
    indexed = {}

    for index, source in enumerate(interval_records):
        if not isinstance(source, dict):
            raise ValueError(f"Interval record {index} must be a mapping.")

        missing = [
            field for field in _REQUIRED_INTERVAL_FIELDS
            if field not in source
        ]
        if missing:
            raise ValueError(
                f"Interval record {index} missing fields: {missing}"
            )

        method = _required_text(source["method"], "method")
        cycle_id = _positive_integer(source["cycle_id"], "cycle_id")
        origin = _required_text(source["origin"], "origin")
        target = _required_text(source["target"], "target")
        horizon = _positive_integer(
            source["horizon_months"], "horizon_months"
        )
        nominal = _finite_number(source["nominal_level"], "nominal_level")
        lower = _finite_number(source["lower"], "lower")
        upper = _finite_number(source["upper"], "upper")

        if nominal not in _NOMINAL_LEVELS:
            raise ValueError(
                f"Unsupported nominal level: {nominal}"
            )

        if lower > upper:
            raise ValueError("Interval lower bound exceeds upper bound.")

        key = (method, cycle_id, origin, target, horizon)

        if key not in forecasts:
            raise ValueError(
                f"Interval has no matching forecast record: {key}"
            )

        interval_key = key + (nominal,)

        if interval_key in indexed:
            raise ValueError(
                f"Duplicate interval record key: {interval_key}"
            )

        indexed[interval_key] = {
            "lower": lower,
            "upper": upper,
            "nominal_level": nominal,
        }

    return indexed


def summarize_r26_calibration(
    forecast_records,
    interval_records=(),
):
    """Summarize R26 evidence without asserting validated coverage."""

    forecasts = _validate_forecast_records(forecast_records)
    intervals = _validate_interval_records(
        interval_records, forecasts
    )

    groups = defaultdict(list)

    for key, row in forecasts.items():
        groups[(row["method"], row["horizon_months"])].append(
            (key, row)
        )

    interval_groups = defaultdict(list)

    for interval_key, interval in intervals.items():
        method, cycle_id, origin, target, horizon, nominal = interval_key
        interval_groups[(method, horizon, nominal)].append(
            (interval_key[:5], interval)
        )

    results = []

    for (method, horizon), entries in sorted(groups.items()):
        errors = np.asarray(
            [row["absolute_error"] for _, row in entries],
            dtype=float,
        )

        quantiles = np.quantile(
            errors,
            _NOMINAL_LEVELS,

"""Independent A1/A2 acceptance evaluation against the A0 reference."""

from collections import defaultdict
from collections.abc import Mapping
import math
import numbers

import numpy as np

from .solarshield_v3_peaks import detect_peak_plateaus


_REQUIRED_FIELDS = (
    "cycle_id",
    "origin",
    "target",
    "prediction",
    "observation",
)

_METRIC_NAMES = (
    "MAE",
    "RMSE",
    "peak_amplitude_error",
    "peak_timing_error",

def _number(value, field):
    """Convert a numeric value and reject missing or non-finite values."""
    if value is None or isinstance(value, (bool, np.bool_)):
        raise ValueError(f"{field}: missing or invalid numeric value.")

    if not isinstance(value, (numbers.Real, str)):
        raise ValueError(f"{field}: value must be numeric.")

    try:
        result = float(value)
    except (TypeError, ValueError, OverflowError) as exc:
        raise ValueError(f"{field}: value cannot be converted to float.") from exc

    if not math.isfinite(result):
        raise ValueError(f"{field}: value must be finite.")

    return result


def _validate_key_value(value, field):
    if value is None or (isinstance(value, str) and not value.strip()):
        raise ValueError(f"{field}: key value is missing.")

    try:
        hash(value)
    except TypeError as exc:
        raise ValueError(f"{field}: key value must be hashable.") from exc

    return value


def _index_records(records, label):
    """Validate and index records using exact cycle/origin/target keys."""
    indexed = {}
    errors = []

    try:
        rows = list(records)
    except TypeError:
        return None, [f"{label}: records must be iterable."]

    if not rows:
        return None, [f"{label}: no records supplied."]

    for position, row in enumerate(rows):
        prefix = f"{label}[{position}]"

        if not isinstance(row, Mapping):
            errors.append(f"{prefix}: record must be a mapping.")
            continue

        missing_fields = [
            field for field in _REQUIRED_FIELDS if field not in row
        ]
        if missing_fields:
            errors.append(
                f"{prefix}: missing fields {missing_fields}."
            )
            continue

        try:
            cycle_id = _validate_key_value(
                row["cycle_id"], f"{prefix}.cycle_id"
            )
            origin = _validate_key_value(
                row["origin"], f"{prefix}.origin"
            )
            target = _validate_key_value(
                row["target"], f"{prefix}.target"
            )

            key = (cycle_id, origin, target)

            if key in indexed:
                errors.append(f"{prefix}: duplicate forecast key {key!r}.")
                continue

            prediction = _number(
                row["prediction"], f"{prefix}.prediction"
            )
            observation = _number(
                row["observation"], f"{prefix}.observation"
            )

            indexed[key] = {
                "cycle_id": cycle_id,
                "origin": origin,
                "target": target,
                "prediction": prediction,
                "observation": observation,
            }

        except ValueError as exc:
            errors.append(str(exc))

    if errors:
        return None, errors

    return indexed, []


def _target_sort_key(value):
    """Provide deterministic ordering for numeric or string targets."""
    if isinstance(value, numbers.Real) and not isinstance(value, bool):
        return (0, float(value))

    if isinstance(value, str):
        return (1, value)

    raise ValueError(
        "Target values must consistently use numeric or string values."
    )


def _phase_hausdorff(reference_phases, candidate_phases):
    """Symmetric Hausdorff distance between global peak phase sets."""
    a = np.asarray(reference_phases, dtype=float)
    b = np.asarray(candidate_phases, dtype=float)

    if a.size == 0 or b.size == 0:
        raise ValueError("Global peak phase sets cannot be empty.")

    distances = np.abs(a[:, None] - b[None, :])

    directed_ab = float(np.max(np.min(distances, axis=1)))
    directed_ba = float(np.max(np.min(distances, axis=0)))

    return max(directed_ab, directed_ba)


def _peak_errors(observations, predictions):
    """Calculate global-maximum amplitude and timing errors for one curve."""
    y = np.asarray(observations, dtype=float)
    p = np.asarray(predictions, dtype=float)

    if y.ndim != 1 or p.ndim != 1 or y.size != p.size:
        raise ValueError("Reference and candidate curves must have equal lengths.")

    if y.size < 2:
        raise ValueError("At least two points are required for peak evaluation.")

    if not np.all(np.isfinite(y)) or not np.all(np.isfinite(p)):
        raise ValueError("Peak evaluation requires finite curve values.")

    reference_peaks = detect_peak_plateaus(y, tolerance=0.0)
    candidate_peaks = detect_peak_plateaus(p, tolerance=0.0)

    reference_phases = [
        peak.center_phase for peak in reference_peaks
    ]
    candidate_phases = [
        peak.center_phase for peak in candidate_peaks
    ]

    amplitude_error = abs(float(np.max(y)) - float(np.max(p)))

    timing_error = _phase_hausdorff(
        reference_phases,
        candidate_phases,
    )

    if not math.isfinite(amplitude_error):
        raise ValueError("Peak amplitude error is non-finite.")

    if not math.isfinite(timing_error):
        raise ValueError("Peak timing error is non-finite.")

    return amplitude_error, timing_error

def evaluate_r32_candidate(
    reference_records,
    candidate_records,
    *,
    candidate_id,
):
    """
    Evaluate one candidate independently against A0.

    Each record must contain:
        cycle_id, origin, target, prediction, observation

    MAE/RMSE use all matched points.
    Peak metrics use the macro-average across cycle-origin curves.
    """

    if candidate_id not in ("A1", "A2"):
        return _unresolved_result(
            candidate_id,
            ["candidate_id must be A1 or A2."],
        )

    reference, reference_errors = _index_records(
        reference_records, "A0"
    )
    candidate, candidate_errors = _index_records(
        candidate_records, candidate_id
    )

    errors = reference_errors + candidate_errors

    if errors:
        return _unresolved_result(candidate_id, errors)

    reference_keys = set(reference)
    candidate_keys = set(candidate)
    matched_keys = reference_keys & candidate_keys
    matched_point_count = len(matched_keys)
    matched_curve_count = len({(key[0], key[1]) for key in matched_keys})

    if reference_keys != candidate_keys:
        missing = sorted(
            (repr(key) for key in reference_keys - candidate_keys)
        )
        unexpected = sorted(
            (repr(key) for key in candidate_keys - reference_keys)
        )

        return _unresolved_result(
            candidate_id,
            [
                "Forecast keys are not identical.",
                f"Missing candidate keys: {missing}",
                f"Unexpected candidate keys: {unexpected}",
            ],
            matched_point_count=matched_point_count,
            matched_curve_count=matched_curve_count,
        )

    ordered_keys = sorted(
        reference_keys,
        key=lambda key: (
            repr(key[0]),
            repr(key[1]),
            _target_sort_key(key[2]),
        ),
    )

    for key in ordered_keys:
        ref_obs = reference[key]["observation"]
        cand_obs = candidate[key]["observation"]

        if not math.isclose(
            ref_obs,
            cand_obs,
            rel_tol=0.0,
            abs_tol=1e-12,
        ):
            return _unresolved_result(
                candidate_id,
                [f"Observation mismatch for forecast key {key!r}."],
                matched_point_count=matched_point_count,
                matched_curve_count=matched_curve_count,
            )

    reference_errors_point = []
    candidate_errors_point = []

    for key in ordered_keys:
        reference_errors_point.append(
            reference[key]["prediction"]
            - reference[key]["observation"]
        )
        candidate_errors_point.append(
            candidate[key]["prediction"]
            - candidate[key]["observation"]
        )

    ref_error = np.asarray(reference_errors_point, dtype=float)
    cand_error = np.asarray(candidate_errors_point, dtype=float)

    ref_mae = float(np.mean(np.abs(ref_error)))
    cand_mae = float(np.mean(np.abs(cand_error)))

    ref_rmse = float(np.sqrt(np.mean(np.square(ref_error))))
    cand_rmse = float(np.sqrt(np.mean(np.square(cand_error))))

    grouped_reference = defaultdict(list)
    grouped_candidate = defaultdict(list)

    for key in ordered_keys:
        group_key = (key[0], key[1])
        grouped_reference[group_key].append(reference[key])
        grouped_candidate[group_key].append(candidate[key])

    reference_peak_amplitude = []
    candidate_peak_amplitude = []
    reference_peak_timing = []
    candidate_peak_timing = []

    try:
        for group_key in sorted(
            grouped_reference,
            key=lambda value: (repr(value[0]), repr(value[1])),
        ):
            ref_rows = grouped_reference[group_key]
            cand_rows = grouped_candidate[group_key]

            ref_rows = sorted(
                ref_rows,
                key=lambda row: _target_sort_key(row["target"]),
            )
            cand_rows = sorted(
                cand_rows,
                key=lambda row: _target_sort_key(row["target"]),
            )

            if len(ref_rows) != len(cand_rows):
                raise ValueError(
                    f"Curve length mismatch for group {group_key!r}."
                )

            observations = [
                row["observation"] for row in ref_rows
            ]
            ref_predictions = [
                row["prediction"] for row in ref_rows
            ]
            cand_predictions = [
                row["prediction"] for row in cand_rows
            ]

            ref_amp, ref_timing = _peak_errors(
                observations, ref_predictions
            )
            cand_amp, cand_timing = _peak_errors(
                observations, cand_predictions
            )

            reference_peak_amplitude.append(ref_amp)
            candidate_peak_amplitude.append(cand_amp)
            reference_peak_timing.append(ref_timing)
            candidate_peak_timing.append(cand_timing)

    except (ValueError, FloatingPointError, OverflowError) as exc:
        return _unresolved_result(
            candidate_id,
            [str(exc)],
            matched_point_count=matched_point_count,
            matched_curve_count=matched_curve_count,
        )

    metrics = {
        "MAE": (
            ref_mae,
            cand_mae,
            cand_mae < ref_mae,
        ),
        "RMSE": (
            ref_rmse,
            cand_rmse,
            cand_rmse <= ref_rmse,
        ),
        "peak_amplitude_error": (
            float(np.mean(reference_peak_amplitude)),
            float(np.mean(candidate_peak_amplitude)),
            float(np.mean(candidate_peak_amplitude))
            <= float(np.mean(reference_peak_amplitude)),
        ),
        "peak_timing_error": (
            float(np.mean(reference_peak_timing)),
            float(np.mean(candidate_peak_timing)),
            float(np.mean(candidate_peak_timing))
            <= float(np.mean(reference_peak_timing)),
        ),
    }

    metric_results = {}
    unresolved_reasons = []

    for name, (ref_value, cand_value, passed) in metrics.items():
        if not math.isfinite(ref_value) or not math.isfinite(cand_value):
            unresolved_reasons.append(
                f"Non-finite metric value: {name}."
            )
            metric_results[name] = {
                "reference": None,
                "candidate": None,
                "status": "UNRESOLVED",
            }
        else:
            metric_results[name] = {
                "reference": ref_value,
                "candidate": cand_value,
                "status": "PASS" if passed else "FAIL",
            }

    statuses = [
        result["status"] for result in metric_results.values()
    ]

    if "UNRESOLVED" in statuses:
        overall_status = "UNRESOLVED"
    elif "FAIL" in statuses:
        overall_status = "FAIL"
    else:
        overall_status = "PASS"

    return {
        "candidate_id": candidate_id,
        "status": overall_status,
        "matched_point_count": len(ordered_keys),
        "matched_curve_count": len(grouped_reference),
        "metrics": metric_results,
        "reasons": unresolved_reasons,
    }

"""Linear-Gaussian Kalman filter for residual-state assimilation."""
from dataclasses import dataclass
import numpy as np


@dataclass(frozen=True)
class KalmanSnapshot:
    state: np.ndarray
    covariance: np.ndarray


class ResidualStateKalmanFilter:
    """State x=[level, slope]. No solar forecast generation is performed."""

    def __init__(
        self,
        transition=((1.0, 1.0), (0.0, 1.0)),
        observation=((1.0, 0.0),),
        process_covariance=((1e-3, 0.0), (0.0, 1e-4)),
        observation_covariance=((1.0,),),
        initial_state=(0.0, 0.0),
        initial_covariance=((1.0, 0.0), (0.0, 1.0)),
    ):
        self.F = np.asarray(transition, dtype=float)
        self.H = np.asarray(observation, dtype=float)
        self.Q = np.asarray(process_covariance, dtype=float)
        self.R = np.asarray(observation_covariance, dtype=float)
        self.x = np.asarray(initial_state, dtype=float).reshape(-1)
        self.P = np.asarray(initial_covariance, dtype=float)

        if not np.all(np.isfinite(self.F)):
            raise ValueError("Transition matrix must be finite.")
        if not np.all(np.isfinite(self.H)):
            raise ValueError("Observation matrix must be finite.")
        if self.H.ndim != 2:
            raise ValueError("Observation matrix must be two-dimensional.")

        n = self.x.size
        m = self.H.shape[0]

        if self.F.shape != (n, n) or self.P.shape != (n, n):
            raise ValueError("Invalid state or transition dimensions.")
        if self.H.shape[1] != n or self.Q.shape != (n, n):
            raise ValueError("Invalid observation or process dimensions.")
        if self.R.shape != (m, m):
            raise ValueError("Invalid observation covariance dimensions.")

        for name, matrix in (
            ("Q", self.Q), ("R", self.R), ("P", self.P)
        ):
            if not np.all(np.isfinite(matrix)):
                raise ValueError(f"{name} must be finite.")
            if not np.allclose(matrix, matrix.T, atol=1e-10):
                raise ValueError(f"{name} must be symmetric.")
            if np.linalg.eigvalsh(matrix).min() < -1e-9:
                raise ValueError(f"{name} must be positive semidefinite.")

        if not np.all(np.isfinite(self.x)):
            raise ValueError("Initial state must be finite.")

    def snapshot(self):
        return KalmanSnapshot(self.x.copy(), self.P.copy())

    def predict(self):
        with np.errstate(over="ignore", invalid="ignore"):
            predicted_state = self.F @ self.x
            predicted_covariance = (
                self.F @ self.P @ self.F.T + self.Q
            )

        predicted_covariance = 0.5 * (
            predicted_covariance + predicted_covariance.T
        )

        if (
            not np.all(np.isfinite(predicted_state))
            or not np.all(np.isfinite(predicted_covariance))
        ):
            raise ValueError("Prediction produced non-finite state or covariance.")

        self.x = predicted_state
        self.P = predicted_covariance
        return self.snapshot()

    def update(self, observation_value):
        z = np.asarray(observation_value, dtype=float).reshape(-1)

        if z.size != self.H.shape[0] or not np.all(np.isfinite(z)):
            raise ValueError("Invalid observation.")

        with np.errstate(over="ignore", invalid="ignore", divide="ignore"):
            innovation = z - self.H @ self.x
            innovation_covariance = (
                self.H @ self.P @ self.H.T + self.R
            )

        if (
            not np.all(np.isfinite(innovation))
            or not np.all(np.isfinite(innovation_covariance))
        ):
            raise ValueError("Non-finite innovation or innovation covariance.")

        try:
            with np.errstate(over="ignore", invalid="ignore", divide="ignore"):
                gain = np.linalg.solve(
                    innovation_covariance, self.H @ self.P
                ).T
        except np.linalg.LinAlgError as exc:
            raise ValueError("Singular innovation covariance.") from exc

        if not np.all(np.isfinite(gain)):
            raise ValueError("Kalman gain is non-finite.")

        with np.errstate(over="ignore", invalid="ignore"):
            updated_state = self.x + gain @ innovation

            # Joseph covariance update for improved numerical stability.
            identity = np.eye(self.x.size)
            residual_map = identity - gain @ self.H
            updated_covariance = (
                residual_map @ self.P @ residual_map.T
                + gain @ self.R @ gain.T
            )

        updated_covariance = 0.5 * (
            updated_covariance + updated_covariance.T
        )

        if (
            not np.all(np.isfinite(updated_state))
            or not np.all(np.isfinite(updated_covariance))
        ):
            raise ValueError("Update produced non-finite state or covariance.")

        self.x = updated_state
        self.P = updated_covariance
        return self.snapshot()

    def assimilate(self, observation_value):
        previous_state = self.x.copy()
        previous_covariance = self.P.copy()

        try:
            self.predict()
            return self.update(observation_value)
        except Exception:
            self.x = previous_state
            self.P = previous_covariance
            raise

"""Stochastic Ensemble Kalman Filter for residual-state assimilation."""
import copy
import numpy as np


class ResidualStateEnsembleKalmanFilter:
    def __init__(
        self,
        ensemble,
        transition=((1.0, 1.0), (0.0, 1.0)),
        observation=((1.0, 0.0),),
        process_covariance=((1e-3, 0.0), (0.0, 1e-4)),
        observation_covariance=((1.0,),),
        seed=0,
    ):
        self.ensemble = np.asarray(ensemble, dtype=float)
        self.F = np.asarray(transition, dtype=float)
        self.H = np.asarray(observation, dtype=float)
        self.Q = np.asarray(process_covariance, dtype=float)
        self.R = np.asarray(observation_covariance, dtype=float)
        self.rng = np.random.default_rng(seed)

        if not np.all(np.isfinite(self.F)):
            raise ValueError("Transition matrix must be finite.")
        if not np.all(np.isfinite(self.H)):
            raise ValueError("Observation matrix must be finite.")
        if self.H.ndim != 2:
            raise ValueError("Observation matrix must be two-dimensional.")

        if self.ensemble.ndim != 2 or self.ensemble.shape[0] < 2:
            raise ValueError("At least two ensemble members are required.")

        if not np.all(np.isfinite(self.ensemble)):
            raise ValueError("Ensemble must be finite.")

        n = self.ensemble.shape[1]
        m = self.H.shape[0]

        if self.F.shape != (n, n):
            raise ValueError("Invalid transition dimensions.")
        if self.H.ndim != 2 or self.H.shape[1] != n:
            raise ValueError("Invalid observation dimensions.")
        if self.Q.shape != (n, n) or self.R.shape != (m, m):
            raise ValueError("Invalid covariance dimensions.")

        for name, matrix in (("Q", self.Q), ("R", self.R)):
            if not np.all(np.isfinite(matrix)):
                raise ValueError(f"{name} must be finite.")
            if not np.allclose(matrix, matrix.T):
                raise ValueError(f"{name} must be symmetric.")
            if np.linalg.eigvalsh(matrix).min() < -1e-9:
                raise ValueError(f"{name} must be positive semidefinite.")

    @property
    def mean(self):
        return self.ensemble.mean(axis=0)

    @property
    def covariance(self):
        return np.atleast_2d(
            np.cov(self.ensemble, rowvar=False, ddof=1)
        )

    def predict(self):
        previous_rng_state = copy.deepcopy(self.rng.bit_generator.state)

        try:
            with np.errstate(over="ignore", invalid="ignore"):
                noise = self.rng.multivariate_normal(
                    np.zeros(self.ensemble.shape[1]),
                    self.Q,
                    size=self.ensemble.shape[0],
                )
                predicted_ensemble = self.ensemble @ self.F.T + noise

            if not np.all(np.isfinite(predicted_ensemble)):
                raise ValueError("Prediction produced a non-finite ensemble.")

            self.ensemble = predicted_ensemble
            return self.ensemble.copy()

        except Exception:
            self.rng.bit_generator.state = previous_rng_state
            raise

    def update(self, observation_value):
        z = np.asarray(observation_value, dtype=float).reshape(-1)

        if z.size != self.H.shape[0] or not np.all(np.isfinite(z)):
            raise ValueError("Invalid observation.")

        previous_rng_state = copy.deepcopy(self.rng.bit_generator.state)

        try:
            with np.errstate(over="ignore", invalid="ignore", divide="ignore"):
                predicted_observations = self.ensemble @ self.H.T

                state_anomalies = (
                    self.ensemble - self.ensemble.mean(axis=0)
                )
                observation_anomalies = (
                    predicted_observations
                    - predicted_observations.mean(axis=0)
                )

                n = self.ensemble.shape[0]

                cross_covariance = (
                    state_anomalies.T @ observation_anomalies / (n - 1)
                )
                observation_covariance = (
                    observation_anomalies.T @ observation_anomalies
                    / (n - 1) + self.R
                )

            intermediate_arrays = (
                predicted_observations,
                state_anomalies,
                observation_anomalies,
                cross_covariance,
                observation_covariance,
            )

            if not all(
                np.all(np.isfinite(array))
                for array in intermediate_arrays
            ):
                raise ValueError("Non-finite EnKF covariance calculation.")

            try:
                with np.errstate(over="ignore", invalid="ignore", divide="ignore"):
                    gain = np.linalg.solve(
                        observation_covariance, cross_covariance.T
                    ).T
            except np.linalg.LinAlgError as exc:
                raise ValueError("Singular ensemble covariance.") from exc

            if not np.all(np.isfinite(gain)):
                raise ValueError("EnKF gain is non-finite.")

            with np.errstate(over="ignore", invalid="ignore"):
                perturbations = self.rng.multivariate_normal(
                    np.zeros(z.size), self.R, size=n
                )
                innovations = (
                    z[None, :] + perturbations - predicted_observations
                )
                updated_ensemble = (
                    self.ensemble + innovations @ gain.T
                )

            if (
                not np.all(np.isfinite(perturbations))
                or not np.all(np.isfinite(innovations))
                or not np.all(np.isfinite(updated_ensemble))
            ):
                raise ValueError("Update produced a non-finite ensemble.")

            self.ensemble = updated_ensemble
            return self.ensemble.copy()

        except Exception:
            self.rng.bit_generator.state = previous_rng_state
            raise

    def assimilate(self, observation_value):
        previous_ensemble = self.ensemble.copy()
        previous_rng_state = copy.deepcopy(self.rng.bit_generator.state)

        try:
            self.predict()
            return self.update(observation_value)
        except Exception:
            self.ensemble = previous_ensemble
            self.rng.bit_generator.state = previous_rng_state
            raise

def sha256_file(path):
    h = hashlib.sha256()
    with path.open("rb") as f:

            h.update(block)
    return h.hexdigest()

HORIZONS = (1, 3, 6, 12)


def require(condition, message):
    if not condition:
        raise ValueError(message)


def month_offset(origin, target):
    return (target.year - origin.year) * 12 + target.month - origin.month


def training_mase_scale(training_values, seasonal_period=1):
    """MASE scale computed from training data only."""
    values = pd.Series(training_values, dtype=float)
    require(len(values) > seasonal_period,
            "Insufficient training values for MASE scale.")
    differences = values.diff(seasonal_period).dropna().abs()
    require(len(differences) > 0, "Empty MASE denominator.")
    scale = float(differences.mean())
    require(pd.notna(scale) and scale > 0,
            "MASE denominator is zero or invalid.")
    return scale


def main():
    print("SolarShield-AI: read-only forecasting protocol validation")
    print(f"Project root: {ROOT}")

    monthly = pd.read_csv(MONTHLY_PATH, parse_dates=["month_start"])

## 6. Residual assimilation

### 6.1 State definition

The residual state is:

\[
x_t =
\begin{bmatrix}
\ell_t\\
s_t
\end{bmatrix}
\]

where \(\ell_t\) represents residual level and \(s_t\) represents residual slope.

The state transition is:

\[
x_t = Fx_{t-1}+w_t
\]

with:

\[
F =
\begin{bmatrix}
1&1\\
0&1
\end{bmatrix},
\qquad
H =
\begin{bmatrix}
1&0
\end{bmatrix}
\]

The observation equation is:

\[
z_t=Hx_t+v_t
\]

where \(z_t\) is the causal A0 forecast residual.

### 6.2 Causal residual construction

Residuals must be generated prequentially:

1. Construct the baseline using information available before the target month.
2. Generate the one-step-ahead baseline prediction.
3. Observe the target month.
4. Calculate the residual.
5. Assimilate that residual into the state.

Future observations must never contribute to an earlier residual or baseline prediction.

The residual-generation procedure must be identical in training and evaluation.

**OPEN:** The initialization and warm-up rule for the earliest months, when sufficient observed history for T2 anchoring is unavailable, requires specification before protocol freeze.

### 6.3 Baseline update and state continuity

At each official forecast snapshot, the current A0 baseline is recomputed using the newly available prefix.

The residual state must not be transferred mechanically between incompatible baseline reference frames.

The preferred method is causal sequential residual assimilation using the same prequential A0 definition at every month. The state is maintained in that residual coordinate system.

A separate replay or state-rebase procedure may only be adopted if synthetic tests establish mathematical equivalence and continuity.

## 7. KF and EnKF

### 7.1 Standard Kalman Filter

Forecast step:

\[
x_t^f=Fx_{t-1}^a
\]

\[
P_t^f=FP_{t-1}^aF^T+Q
\]

Innovation:

\[
\nu_t=z_t-Hx_t^f
\]

\[
S_t=HP_t^fH^T+R
\]

Gain:

\[
K_t=P_t^fH^TS_t^{-1}
\]

State update:

\[
x_t^a=x_t^f+K_t\nu_t
\]

Covariance update must use the numerically stable Joseph form.

### 7.2 EnKF

The EnKF will use an ensemble-based approximation of the state distribution.

Candidate ensemble sizes:

- 100
- 250
- 500

Multiple predeclared random seeds must be evaluated.

The EnKF must use the same information boundary, baseline residual definition and training-only estimation policy as the KF.

EnKF is a candidate for comparison, not an assumed improvement over the linear KF.

## 8. Q/R/P0 estimation

### 8.1 Proposed parameterization

The proposed local-linear-trend process covariance is:

\[
Q=q
\begin{bmatrix}
1/3&1/2\\
1/2&1
\end{bmatrix},
\qquad q>0
\]

Observation covariance:

\[
R=[r],\qquad r>0
\]

Initial covariance:

\[
P_0=
\begin{bmatrix}
p_\ell&0\\
0&p_s
\end{bmatrix},
\qquad p_\ell,p_s>0
\]

Initial state mean:

\[
m_0=
\begin{bmatrix}
0\\
0
\end{bmatrix}
\]

These are proposed estimation structures, not scientifically validated parameter values.

### 8.2 Likelihood objective

Use the Gaussian innovation negative log-likelihood:

\[
\mathcal L(\theta)=
\frac12\sum_t
\left[
\log(2\pi)+\log(S_t)+\frac{\nu_t^2}{S_t}
\right]
\]

where:

\[
\theta=(q,r,p_\ell,p_s)
\]

The exact sequence aggregation, initialization, parameter bounds and optimizer configuration must be documented before freeze.

### 8.3 Estimation safeguards

- Estimate parameters using training-only residual sequences.
- Use positive parameter constraints.
- Record optimizer, initialization, bounds and convergence.
- Record boundary hits and identifiability diagnostics.
- Reject non-finite or singular solutions.
- Never silently substitute synthetic-test defaults for failed estimates.
- Do not use calibration or final evaluation observations to estimate Q, R or P0.

**OPEN:** Parameter identifiability, regularization, minimum residual-history requirements and optimizer acceptance criteria require technical review.

## 10. DTW and shape-aware evaluation

DTW must evaluate cycle-shape similarity without allowing temporal alignment to hide important physical differences.

Required outputs:

- Raw DTW distance.
- Normalized DTW distance.
- Peak timing error.
- Peak phase displacement.
- Peak amplitude error.
- Peak count and ambiguity.

2. Distance function.

4. Path constraints.

6. Missing-value handling.

Phase-aware warping is preferred over an unqualified integer-index window.

**OPEN:** Exact warping window, normalization, path-length correction and composite metric weights must be selected using cycles 6â€“12 and frozen before calibration.

## 11. Peak evaluation

Peak evaluation must not be limited to the global maximum.

The method must identify relevant local peaks and support:

- Multiple peaks.
- Plateau-shaped peaks.
- Peak matching between observed and predicted curves.
- Peak timing and phase displacement.
- Absolute and relative amplitude error.
- Missing and additional predicted peaks.

## 2. R21 — Owner-approved Option A parameterization

The estimator uses shared physical variance parameters q and r across eligible training residual sequences.

For each eligible sequence i, let s_i be its training-only robust residual scale.

The robust scale is calculated as:

- s_i = 1.4826 × MAD(z_i)
- If MAD is zero or invalid, evaluate IQR(z_i) / 1.349.
- If both are zero or invalid, fail with UNRESOLVED_RESIDUAL_SCALE.

Residuals are normalized using their own training-only scale.

The authoritative process-noise covariance shape is:

Q_shape = [[1, 1/2], [1/2, 1]]

The physical process-noise covariance is:

Q_physical = q × Q_shape

For each normalized sequence:

Q_i_normalized = (q / s_i²) × Q_shape

The observation-noise covariance is:

R_physical = [[r]]

R_i_normalized = [[r / s_i²]]

The initial covariance is:

P0 = diag(1, 1/36)

Each residual sequence is initialized independently using the specified initial state and covariance.

The sequence likelihood includes the physical-scale Jacobian contribution:

valid_observation_count_i × log(s_i)

The objective is the sum of the sequence negative log-likelihood contributions.

The earlier conflicting Q shape in Draft 1.3, q × [[1/3, 1/2], [1/2, 1]], is superseded for R21 by the owner-approved matrix in this section.

## 3. R21 optimization settings

- Parameters: log(q), log(r)
- Bounds: [-13.8155, 4.6052]
- Initializations: all nine combinations of q and r in {0.0001, 0.01, 1.0}
- Maximum iterations: 10000
- ftol: 1e-9
- gtol: 1e-6

The implementation must report optimizer success, boundary hits, initialization sensitivity, numerical failures, and local curvature diagnostics.

Minimum-history eligibility remains three completed training-cycle residual sequences, each containing at least twelve valid post-warm-up observations.
