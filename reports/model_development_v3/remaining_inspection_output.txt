from pathlib import Path
import hashlib

import numpy as np
import pandas as pd


PHASE_POINTS = 121
COMPLETED_CYCLES = tuple(range(1, 25))


class CyclePreparationError(ValueError):
    """Raised when shared source data cannot be safely interpreted."""


def sha256_file(path):
    digest = hashlib.sha256()
    with Path(path).open("rb") as handle:
        for block in iter(lambda: handle.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def month_diff(start, end):
    return (end.year - start.year) * 12 + end.month - start.month


def prepare_cycle_curves(monthly_path, cycle_path, phase_points=PHASE_POINTS):
    """
    Prepare completed solar-cycle curves using one shared V2 convention.

    Returns:
        curves: {cycle_number: 1D numpy array}
        durations: {cycle_number: float}
        raw_peaks: {cycle_number: float}
        raw_peak_indices: {cycle_number: int}
        source_errors: {cycle_number: reason}
        phase_grid: 1D numpy array
        monthly_sha256, cycle_sha256

    Invalid individual cycles are recorded in source_errors.
    Invalid global source structure raises CyclePreparationError.
    No model fitting or experiment output is performed.
    """
    if not isinstance(phase_points, int) or phase_points < 2:
        raise CyclePreparationError("phase_points must be an integer >= 2.")

    monthly_path = Path(monthly_path)
    cycle_path = Path(cycle_path)

    if not monthly_path.is_file():
        raise CyclePreparationError(f"Monthly dataset not found: {monthly_path}")
    if not cycle_path.is_file():
        raise CyclePreparationError(f"Cycle summary not found: {cycle_path}")

    monthly_hash = sha256_file(monthly_path)
    cycle_hash = sha256_file(cycle_path)

    monthly = pd.read_csv(monthly_path)
    cycles = pd.read_csv(cycle_path)

    required_monthly = {"month_start", "ssn"}
    required_cycles = {
        "cycle_number",
        "status",
        "published_minimum_month",
        "months_expected_min_to_min",
    }

    missing_monthly = required_monthly - set(monthly.columns)
    missing_cycles = required_cycles - set(cycles.columns)

    if missing_monthly:
        raise CyclePreparationError(
            f"Missing monthly columns: {sorted(missing_monthly)}"
        )
    if missing_cycles:
        raise CyclePreparationError(
            f"Missing cycle columns: {sorted(missing_cycles)}"
        )

    monthly["month_start"] = pd.to_datetime(
        monthly["month_start"], errors="coerce"
    )
    monthly["ssn"] = pd.to_numeric(monthly["ssn"], errors="coerce")

    if monthly["month_start"].isna().any():
        raise CyclePreparationError("Monthly dataset contains invalid dates.")

    if not monthly["month_start"].is_unique:
        raise CyclePreparationError("Monthly dataset contains duplicate dates.")

    if not monthly["month_start"].eq(
        monthly["month_start"].dt.to_period("M").dt.to_timestamp()
    ).all():
        raise CyclePreparationError(
            "Monthly dates must be normalized to month-start timestamps."
        )

    monthly = monthly.sort_values("month_start").reset_index(drop=True)

    cycles["cycle_number"] = pd.to_numeric(
        cycles["cycle_number"], errors="coerce"
    )
    cycles["published_minimum_month"] = pd.to_datetime(
        cycles["published_minimum_month"], errors="coerce"
    )
    cycles["months_expected_min_to_min"] = pd.to_numeric(
        cycles["months_expected_min_to_min"], errors="coerce"
    )

    if cycles["cycle_number"].isna().any():
        raise CyclePreparationError("Cycle summary contains invalid cycle numbers.")

    cycle_numbers = cycles["cycle_number"].to_numpy(dtype=float)

    if not np.isfinite(cycle_numbers).all():
        raise CyclePreparationError(
            "Cycle numbers must all be finite."
        )

    if not np.equal(cycle_numbers, np.floor(cycle_numbers)).all():
        raise CyclePreparationError(
            "Cycle numbers must all be integers."
        )

    if cycles["cycle_number"].duplicated().any():
        raise CyclePreparationError("Cycle summary contains duplicate cycle numbers.")

    cycle_rows = {
        int(row["cycle_number"]): row
        for _, row in cycles.iterrows()
    }

    if 25 not in cycle_rows:
        raise CyclePreparationError("Cycle 25 row is required for Cycle 24 boundary.")

    cycle25 = cycle_rows[25]
    next_minimum = cycle25["published_minimum_month"]

    if pd.isna(next_minimum):
        raise CyclePreparationError(
            "Cycle 25 published minimum is required as Cycle 24 boundary."
        )

    starts = {}
    source_errors = {}

    for cycle in COMPLETED_CYCLES:
        row = cycle_rows.get(cycle)

        if row is None:
            source_errors[cycle] = "missing_cycle_summary_row"
            continue

        # Preserve a valid published minimum as a boundary independently
        # of whether this cycle is eligible as a training/heldout curve.
        if pd.isna(row["published_minimum_month"]):
            source_errors[cycle] = "missing_published_minimum"
            continue

        starts[cycle] = row["published_minimum_month"]

        if str(row["status"]).strip().lower() != "completed":
            source_errors[cycle] = "cycle_not_marked_completed"
            continue

        if pd.isna(row["months_expected_min_to_min"]):
            source_errors[cycle] = "missing_expected_duration"
            continue

        duration = float(row["months_expected_min_to_min"])
        if not np.isfinite(duration) or not duration.is_integer() or duration < 2:
            source_errors[cycle] = "invalid_expected_duration"
            continue

    starts[25] = next_minimum

    date_index = monthly.set_index("month_start")
    phase_grid = np.linspace(0.0, 1.0, phase_points)

    curves = {}
    durations = {}
    raw_peaks = {}
    raw_peak_indices = {}
    raw_peak_months = {}
    official_smoothed_max_strengths = {}
    official_smoothed_max_months = {}

    for cycle in COMPLETED_CYCLES:
        if cycle in source_errors:
            continue

        if cycle not in starts or (cycle + 1) not in starts:
            source_errors[cycle] = "missing_cycle_boundary"
            continue

        start = starts[cycle]
        end = starts[cycle + 1]
        row = cycle_rows[cycle]
        expected_duration = int(row["months_expected_min_to_min"])

        if pd.isna(start) or pd.isna(end) or end <= start:
            source_errors[cycle] = "invalid_cycle_boundaries"
            continue

        if month_diff(start, end) != expected_duration:
            source_errors[cycle] = "boundary_duration_mismatch"
            continue

        expected_months = pd.date_range(
            start, end, freq="MS", inclusive="left"
        )

        segment = monthly.loc[
            (monthly["month_start"] >= start)
            & (monthly["month_start"] < end)
        ].sort_values("month_start")

        if segment["month_start"].tolist() != list(expected_months):
            source_errors[cycle] = "missing_or_extra_monthly_rows"
            continue

        if len(segment) != expected_duration:
            source_errors[cycle] = "observed_duration_mismatch"
            continue

        values = segment["ssn"].to_numpy(dtype=float)

        if not np.isfinite(values).all():
            source_errors[cycle] = "non_finite_monthly_ssn"
            continue

        source_phase = np.linspace(0.0, 1.0, len(values))
        curves[cycle] = np.interp(phase_grid, source_phase, values)

        peak_index = int(np.argmax(values))
        durations[cycle] = float(expected_duration)
        raw_peaks[cycle] = float(values[peak_index])
        raw_peak_indices[cycle] = peak_index
        raw_peak_months[cycle] = (
            pd.Timestamp(segment.iloc[peak_index]["month_start"]).strftime("%Y-%m")
        )

        official_strength = row.get("official_smoothed_max_sn")
        official_month = row.get("official_smoothed_max_month")

        official_smoothed_max_strengths[cycle] = (
            None if pd.isna(official_strength) else float(official_strength)
        )
        official_smoothed_max_months[cycle] = (
            None
            if pd.isna(official_month) or not str(official_month).strip()
            else str(official_month)[:7]
        )

    return {
        "curves": curves,
        "durations": durations,
        "raw_peaks": raw_peaks,
        "raw_peak_indices": raw_peak_indices,
        "raw_peak_months": raw_peak_months,
        "official_smoothed_max_strengths": official_smoothed_max_strengths,
        "official_smoothed_max_months": official_smoothed_max_months,
        "source_errors": source_errors,
        "phase_grid": phase_grid,
        "phase_points": phase_points,
        "monthly_sha256": monthly_hash,
        "cycle_sha256": cycle_hash,
        "cycle_boundary_convention": (
            "Published minimum inclusive; next published minimum exclusive. "
            "Cycle 25 minimum is used only as Cycle 24 end boundary."
        ),
        "phase_convention": (
            "First observed month maps to phase 0; last observed month "
            "before the next minimum maps to phase 1; linear interpolation."
        ),
    }

"""Read-only loaders for SolarShield-AI raw data.

-1 is a "no value" sentinel in some columns (sunspot counts and indices cannot
be negative). These loaders turn -1 into NaN IN MEMORY ONLY, in explicitly
listed columns, and report how many values were masked. Files under data/ are
never written. Nothing is imputed.
"""
from __future__ import annotations
import json
from pathlib import Path
import numpy as np
import pandas as pd

ROOT = Path(__file__).resolve().parent.parent
RAW = ROOT / "data" / "raw"
SENTINEL = -1

SILSO_COLS = ["year", "month", "decimal_year", "ssn", "ssn_std", "n_obs", "definitive"]
SILSO_SENTINEL_COLS = ["ssn_std", "n_obs"]
NOAA_OBSERVED_SENTINEL_COLS = ["smoothed_ssn", "observed_swpc_ssn", "smoothed_swpc_ssn", "f10.7", "smoothed_f10.7"]


def mask_sentinels(df: pd.DataFrame, columns, sentinel=SENTINEL):
    """Return (copy with sentinel -> NaN in `columns`, {column: count masked})."""
    out = df.copy()
    counts = {}
    for c in columns:
        m = out[c] == sentinel
        counts[c] = int(m.sum())
        out[c] = out[c].astype("float64").mask(m, np.nan)
    return out, counts


def load_silso(raw_dir: Path = RAW):
    df = pd.read_csv(raw_dir / "SILSO_Monthly_Sunspot_Data.csv", sep=";", header=None,
                     names=SILSO_COLS, skipinitialspace=True)
    df, counts = mask_sentinels(df, SILSO_SENTINEL_COLS)
    df["month_start"] = pd.to_datetime(dict(year=df["year"], month=df["month"], day=1))
    return df, counts


def load_noaa_observed(raw_dir: Path = RAW):
    with open(raw_dir / "NOAA_Observed_Solar_Cycle.json", encoding="utf-8") as fh:
        df = pd.DataFrame(json.load(fh))
    df, counts = mask_sentinels(df, NOAA_OBSERVED_SENTINEL_COLS)
    df["month_start"] = pd.to_datetime(df["time-tag"] + "-01")
    return df, counts


def load_solar_cycle_reference(root: Path = ROOT):
    """Standardized cycle table; dates stay month-precision strings (YYYY-MM), blanks -> NA."""
    return pd.read_csv(root / "data" / "metadata" / "solar_cycle_reference.csv",
                       dtype=str, keep_default_na=False).replace("", pd.NA)

from pathlib import Path
from datetime import datetime, timezone

import json
import platform

DATA_PATH = ROOT / "data" / "processed" / "sunspot_monthly_clean.csv"
START = pd.Timestamp("1976-01-01")
LAST_ORIGIN = pd.Timestamp("2025-06-01")
CUTOFF = pd.Timestamp("2026-06-01")
HORIZONS = (1, 3, 6, 12)


def sha256_file(path):
    h = hashlib.sha256()
    with path.open("rb") as f:
        for block in iter(lambda: f.read(1024 * 1024), b""):
            h.update(block)
    return h.hexdigest()


def parse_bool(series):
    result = series.astype(str).str.lower().map({
        "true": True, "false": False, "1": True, "0": False
    })
    if result.isna().any():
        raise ValueError("Unrecognized is_provisional value.")
    return result.astype(bool)


def main():
    if not DATA_PATH.is_file():
        raise FileNotFoundError(DATA_PATH)

    dataset_hash = sha256_file(DATA_PATH)
    data = pd.read_csv(DATA_PATH, parse_dates=["month_start"])

    required = {"month_start", "ssn", "is_provisional"}
    if not required.issubset(data.columns):
        raise ValueError(f"Missing columns: {required - set(data.columns)}")
    if data["month_start"].isna().any() or data["month_start"].duplicated().any():
        raise ValueError("Missing or duplicate month_start values.")

    data = data.sort_values("month_start").reset_index(drop=True)
    if data["ssn"].isna().any():
        raise ValueError("Missing SSN values.")
    data["is_provisional"] = parse_bool(data["is_provisional"])

    expected = pd.date_range(
        data["month_start"].iloc[0], data["month_start"].iloc[-1], freq="MS"
    )
    if data["month_start"].tolist() != list(expected):
        raise ValueError("Monthly dates are not continuous month-start dates.")

    indexed = data.set_index("month_start", drop=False)
    origins = pd.date_range(START, LAST_ORIGIN, freq="MS")
    if not origins.isin(indexed.index).all():
        raise ValueError("One or more forecast origins are missing.")

    records = []
    excluded = []
    for origin in origins:
        training = data.loc[data["month_start"] <= origin, "ssn"].astype(float)
        if len(training) < 2:
            raise ValueError(f"Insufficient training data at {origin:%Y-%m}")

        # Non-seasonal MASE denominator: mean absolute first difference
        # calculated strictly from observations available through origin.
        mase_scale = float(training.diff().abs().dropna().mean())
        if not np.isfinite(mase_scale) or mase_scale <= 0:
            raise ValueError(f"Invalid MASE scale at {origin:%Y-%m}")

        for horizon in HORIZONS:
            target = origin + pd.DateOffset(months=horizon)
            if target > CUTOFF:
                excluded.append({
                    "origin": origin, "horizon": horizon,
                    "target_month": target, "reason": "beyond_cutoff"
                })
                continue
            if target not in indexed.index:
                excluded.append({
                    "origin": origin, "horizon": horizon,
                    "target_month": target, "reason": "target_missing"
                })
                continue

            target_row = indexed.loc[target]
            if bool(target_row["is_provisional"]):
                excluded.append({
                    "origin": origin, "horizon": horizon,
                    "target_month": target, "reason": "provisional_target"
                })
                continue

            persistence_source = origin
            seasonal_source = target - pd.DateOffset(months=12)
            for source in (persistence_source, seasonal_source):
                if source > origin:
                    raise ValueError(
                        f"Leakage: source {source} exceeds origin {origin}"
                    )
                if source not in indexed.index:
                    raise ValueError(f"Missing baseline source month: {source}")

            actual = float(target_row["ssn"])
            predictions = {
                "persistence": (
                    float(indexed.loc[persistence_source, "ssn"]),
                    persistence_source
                ),
                "seasonal_naive_12": (
                    float(indexed.loc[seasonal_source, "ssn"]),
                    seasonal_source
                ),
            }

            for model, (prediction, source_month) in predictions.items():
                if not np.isfinite(prediction):
                    raise ValueError("Non-finite baseline prediction.")
                records.append({
                    "origin": origin,
                    "training_start": data["month_start"].iloc[0],
                    "training_cutoff": origin,
                    "training_size": len(training),
                    "target_month": target,
                    "horizon_months": horizon,
                    "model": model,
                    "prediction": prediction,
                    "actual_ssn": actual,
                    "source_month": source_month,
                    "source_ssn": prediction,
                    "absolute_error": abs(actual - prediction),
                    "squared_error": (actual - prediction) ** 2,
                    "mase_scale_training_only": mase_scale,
                    "scaled_absolute_error": abs(actual - prediction) / mase_scale,
                    "dataset_sha256": dataset_hash,
                })

    forecasts = pd.DataFrame(records)
    if forecasts.empty:
        raise ValueError("No eligible forecasts were generated.")

    # Check the two baselines coincide at the 12-month horizon.
    h12 = forecasts.loc[forecasts["horizon_months"] == 12]
    pivot = h12.pivot(index="origin", columns="model", values="prediction")
    if not np.allclose(
        pivot["persistence"].to_numpy(),
        pivot["seasonal_naive_12"].to_numpy(),
    ):
        raise ValueError("12-month baseline equality check failed.")

    summary_rows = []
    for (horizon, model), group in forecasts.groupby(
        ["horizon_months", "model"], sort=True
    ):
        summary_rows.append({
            "horizon_months": int(horizon),
            "model": model,
            "n_forecasts": len(group),
            "MAE": float(group["absolute_error"].mean()),
            "RMSE": float(np.sqrt(group["squared_error"].mean())),
            "MASE": float(group["scaled_absolute_error"].mean()),
            "mean_training_size": float(group["training_size"].mean()),
        })
    summary = pd.DataFrame(summary_rows)

    # Positive skill means lower MAE than the named reference baseline.
    mae = summary.pivot(index="horizon_months", columns="model", values="MAE")
    skill_rows = []
    for horizon in mae.index:
        p_mae = float(mae.loc[horizon, "persistence"])
        s_mae = float(mae.loc[horizon, "seasonal_naive_12"])
        for model in ("persistence", "seasonal_naive_12"):
            model_mae = float(mae.loc[horizon, model])
            skill_rows.append({
                "horizon_months": int(horizon),
                "model": model,
                "skill_vs_persistence": (
                    1.0 - model_mae / p_mae if p_mae > 0 else np.nan
                ),
                "skill_vs_seasonal_naive_12": (
                    1.0 - model_mae / s_mae if s_mae > 0 else np.nan
                ),
            })
    skill = pd.DataFrame(skill_rows)

    stamp = datetime.now().strftime("%Y%m%d_%H%M%S")
    outdir = ROOT / "reports" / "development" / "v1_1" / f"task1_monthly_{stamp}"
    outdir.mkdir(parents=True, exist_ok=False)

    forecasts.to_csv(outdir / "forecast_records.csv", index=False)
    summary.to_csv(outdir / "metrics_by_horizon.csv", index=False)
    skill.to_csv(outdir / "baseline_relative_skill.csv", index=False)
    pd.DataFrame(excluded).to_csv(outdir / "excluded_targets.csv", index=False)

    metadata = {
        "protocol": "SolarShield-AI Forecasting Strategy v1.1",
        "task": "Task 1 monthly baseline hindcasts",
        "dataset_path": str(DATA_PATH.relative_to(ROOT)),
        "dataset_sha256": dataset_hash,
        "source_version_note": (
            "Current processed SILSO dataset; archived historical vintages "
            "not established. Results are retrospective hindcasts."
        ),
        "primary_cutoff": str(CUTOFF.date()),
        "first_origin": str(START.date()),
        "last_origin": str(LAST_ORIGIN.date()),
        "horizons_months": list(HORIZONS),
        "training_rule": "Expanding window, observations through origin inclusive",
        "target_rule": "Targets after cutoff or provisional targets excluded",
        "mase_definition": (
            "Mean absolute first difference of training SSN through each origin; "
            "per-forecast scaled absolute errors averaged by horizon and model"
        ),
        "models": {
            "persistence": "Forecast equals SSN at forecast origin",
            "seasonal_naive_12": "Forecast equals SSN 12 calendar months before target",
        },
        "software": {
            "python": sys.version,
            "platform": platform.platform(),
            "pandas": pd.__version__,
            "numpy": np.__version__,
        },
        "created_utc": datetime.now(timezone.utc).isoformat(),
        "forecast_records": len(forecasts),
        "excluded_target_records": len(excluded),
        "outputs": [
            "forecast_records.csv", "metrics_by_horizon.csv",
            "baseline_relative_skill.csv", "excluded_targets.csv"
        ],
        "limitations": [
            "Overlapping-origin errors are dependent.",
            "Historical data revisions prevent strict real-time simulation.",
            "Baselines are comparators, not evidence of model superiority."
        ],
    }
    (outdir / "run_metadata.json").write_text(
        json.dumps(metadata, indent=2), encoding="utf-8"
    )

    print("PASS: Task 1 baseline hindcasts completed.")
    print(f"Output directory: {outdir}")
    print(f"Forecast records: {len(forecasts)}")
    print(f"Excluded target records: {len(excluded)}")
    print("\nMetrics by horizon:")
    print(summary.to_string(index=False))
    print("\nBaseline-relative skill:")
    print(skill.to_string(index=False))
    print("\n12-month baseline equality: PASS")
    print("No old selection artifacts were read or written by this runner.")


if __name__ == "__main__":
    try:

def sha256_file(path):

    with path.open("rb") as f:
        for block in iter(lambda: f.read(1024 * 1024), b""):
            h.update(block)
    return h.hexdigest()


def main():
    if not CYCLE_PATH.is_file():
        raise FileNotFoundError(CYCLE_PATH)

    source_hash = sha256_file(CYCLE_PATH)
    df = pd.read_csv(CYCLE_PATH)

    required = {
        "cycle_number", "status", "published_minimum_month",
        "official_smoothed_max_month", "official_smoothed_max_sn",
        "months_expected_min_to_min",
    }
    if not required.issubset(df.columns):
        raise ValueError(f"Missing columns: {required - set(df.columns)}")

    df["cycle_number"] = pd.to_numeric(df["cycle_number"], errors="raise")
    df["published_minimum_month"] = pd.to_datetime(

    )
    df["official_smoothed_max_month"] = pd.to_datetime(
        df["official_smoothed_max_month"], errors="coerce"
    )

    complete = df.loc[
        df["cycle_number"].between(1, 24)
        & df["status"].eq("completed")
    ].copy().sort_values("cycle_number")

    if complete["cycle_number"].astype(int).tolist() != list(range(1, 25)):
        raise ValueError("Expected exactly completed Cycles 1-24.")

    if complete[
        [

            "official_smoothed_max_sn", "months_expected_min_to_min"
        ]
    ].isna().any().any():
        raise ValueError("Missing published labels in completed cycles.")

    complete["peak_strength"] = complete["official_smoothed_max_sn"].astype(float)
    complete["peak_timing_months"] = (
        (complete["official_smoothed_max_month"].dt.year
         - complete["published_minimum_month"].dt.year) * 12
        + complete["official_smoothed_max_month"].dt.month
        - complete["published_minimum_month"].dt.month
    ).astype(float)
    complete["duration_months"] = complete["months_expected_min_to_min"].astype(float)

    if (complete["peak_timing_months"] < 0).any():
        raise ValueError("Peak month precedes the published cycle minimum.")
    if (complete["duration_months"] <= 0).any():
        raise ValueError("Non-positive cycle duration.")

    records = []
    for heldout in range(13, 25):
        train = complete.loc[complete["cycle_number"] < heldout]
        test = complete.loc[complete["cycle_number"] == heldout]
        if len(test) != 1 or len(train) != heldout - 1:
            raise ValueError(f"Invalid fold for Cycle {heldout}.")

        actual_row = test.iloc[0]
        for target in TARGETS:
            values = train[target].astype(float).to_numpy()
            actual = float(actual_row[target])
            prediction = float(np.median(values))
            lower = float(np.quantile(values, 0.10))
            upper = float(np.quantile(values, 0.90))

            records.append({
                "heldout_cycle": heldout,
                "training_cycles_start": int(train["cycle_number"].min()),
                "training_cycles_end": int(train["cycle_number"].max()),
                "training_cycle_count": len(train),
                "target": target,
                "actual": actual,
                "prediction_median": prediction,
                "absolute_error": abs(actual - prediction),
                "squared_error": (actual - prediction) ** 2,
                "interval_lower_q10": lower,
                "interval_upper_q90": upper,
                "interval_width": upper - lower,
                "interval_covered": bool(lower <= actual <= upper),
                "training_cycle_values_sha256": hashlib.sha256(
                    np.asarray(values, dtype="<f8").tobytes()
                ).hexdigest(),
                "cycle_summary_sha256": source_hash,
            })

    forecasts = pd.DataFrame(records)
    metrics = []
    for target, group in forecasts.groupby("target", sort=False):
        metrics.append({
            "target": target,
            "n_heldout_cycles": len(group),
            "MAE": float(group["absolute_error"].mean()),
            "RMSE": float(np.sqrt(group["squared_error"].mean())),
            "q10_q90_empirical_coverage": float(group["interval_covered"].mean()),
            "mean_interval_width": float(group["interval_width"].mean()),
        })
    metrics = pd.DataFrame(metrics)

    stamp = datetime.now().strftime("%Y%m%d_%H%M%S")
    outdir = ROOT / "reports" / "development" / "v1_1" / f"task2_cycle_analogs_{stamp}"
    outdir.mkdir(parents=True, exist_ok=False)

    metrics.to_csv(outdir / "cycle_hindcast_metrics.csv", index=False)

        "timing_definition": "Elapsed calendar months from published minimum to published smoothed maximum",

        "interval_definition": "Training-only empirical 10th and 90th percentiles; descriptive, not assumed calibrated",
        "cycle_summary_sha256": source_hash,

            "python": sys.version,

    (outdir / "run_metadata.json").write_text(

from pathlib import Path
import sys

import pandas as pd

ROOT = Path(__file__).resolve().parents[2]
MONTHLY_PATH = ROOT / "data" / "processed" / "sunspot_monthly_clean.csv"
CYCLE_PATH = ROOT / "data" / "processed" / "solar_cycle_summary.csv"

START = pd.Timestamp("1976-01-01")
LAST_ORIGIN = pd.Timestamp("2025-06-01")
CUTOFF = pd.Timestamp("2026-06-01")
HORIZONS = (1, 3, 6, 12)


def require(condition, message):
    if not condition:
        raise ValueError(message)


def month_offset(origin, target):
    return (target.year - origin.year) * 12 + target.month - origin.month


def training_mase_scale(training_values, seasonal_period=1):
    """MASE scale computed from training data only."""
    values = pd.Series(training_values, dtype=float)
    require(len(values) > seasonal_period,
            "Insufficient training values for MASE scale.")
    differences = values.diff(seasonal_period).dropna().abs()
    require(len(differences) > 0, "Empty MASE denominator.")
    scale = float(differences.mean())
    require(pd.notna(scale) and scale > 0,
            "MASE denominator is zero or invalid.")
    return scale


def main():
    print("SolarShield-AI: read-only forecasting protocol validation")
    print(f"Project root: {ROOT}")

    monthly = pd.read_csv(MONTHLY_PATH, parse_dates=["month_start"])
    cycles = pd.read_csv(CYCLE_PATH)

    required_monthly = {
        "month_start", "ssn", "is_provisional", "cycle_number"
    }
    require(required_monthly.issubset(monthly.columns),
            f"Missing monthly columns: {required_monthly - set(monthly.columns)}")
    require(monthly["month_start"].notna().all(), "Missing month_start.")
    require(monthly["month_start"].is_unique, "Duplicate month_start values.")
    require(monthly["ssn"].notna().all(), "Missing SSN values.")

    monthly = monthly.sort_values("month_start").reset_index(drop=True)
    expected_dates = pd.date_range(
        monthly["month_start"].iloc[0],
        monthly["month_start"].iloc[-1],
        freq="MS",
    )
    require(
        monthly["month_start"].tolist() == list(expected_dates),
        "Monthly dates are not a continuous month-start sequence.",
    )

    monthly["is_provisional"] = (
        monthly["is_provisional"].astype(str).str.lower().map(
            {"true": True, "false": False, "1": True, "0": False}
        )
    )
    require(monthly["is_provisional"].notna().all(),
            "Unrecognized is_provisional values.")

    indexed = monthly.set_index("month_start", drop=False)
    origins = pd.date_range(START, LAST_ORIGIN, freq="MS")
    require(origins.isin(indexed.index).all(),
            "One or more planned forecast origins are absent.")

    print(f"Monthly rows: {len(monthly)}")
    print(f"Dataset range: {monthly['month_start'].min():%Y-%m} to "
          f"{monthly['month_start'].max():%Y-%m}")
    print(f"Planned monthly origins: {len(origins)}")

    print("\nMonthly target and leakage checks")
    for horizon in HORIZONS:
        eligible = 0
        provisional = 0
        missing = 0
        beyond_cutoff = 0
        source_errors = 0

        for origin in origins:
            target = origin + pd.DateOffset(months=horizon)

            if month_offset(origin, target) != horizon or target <= origin:
                source_errors += 1
                continue

            if target > CUTOFF:
                beyond_cutoff += 1
                continue

            if target not in indexed.index:
                missing += 1
                continue

            target_row = indexed.loc[target]
            if bool(target_row["is_provisional"]):
                provisional += 1
                continue

            eligible += 1

            # Validate baseline predictor dates against forecast origin.
            persistence_source = origin
            seasonal_source = target - pd.DateOffset(months=12)

            if persistence_source > origin or seasonal_source > origin:
                source_errors += 1
            if persistence_source not in indexed.index:
                source_errors += 1
            if seasonal_source not in indexed.index:
                source_errors += 1

        require(source_errors == 0,
                f"Leakage or alignment errors found at horizon {horizon}.")
        print(
            f"h={horizon:2d}: eligible={eligible}, "
            f"provisional={provisional}, missing={missing}, "
            f"beyond_cutoff={beyond_cutoff}, source_errors={source_errors}"
        )

    # Verify MASE scaling can be calculated from a training prefix only.
    example_origin = pd.Timestamp("2000-01-01")
    training = monthly.loc[
        monthly["month_start"] <= example_origin, "ssn"
    ]
    scale = training_mase_scale(training)
    require(pd.notna(scale) and scale > 0, "Invalid example MASE scale.")
    print(
        f"\nTraining-only MASE scale check: PASS "
        f"(example cutoff={example_origin:%Y-%m}, "
        f"training_rows={len(training)})"
    )

    required_cycle = {
        "cycle_number", "status", "published_minimum_month",
        "official_smoothed_max_month", "official_smoothed_max_sn",
        "months_expected_min_to_min",
    }
    require(required_cycle.issubset(cycles.columns),
            f"Missing cycle columns: {required_cycle - set(cycles.columns)}")

    complete = cycles.loc[
        cycles["cycle_number"].between(1, 24)
        & cycles["status"].eq("completed")
    ].copy().sort_values("cycle_number")

    require(
        complete["cycle_number"].astype(int).tolist() == list(range(1, 25)),
        "Completed-cycle labels are not exactly Cycles 1-24.",
    )

    for col in (
        "published_minimum_month",
        "official_smoothed_max_month",
        "official_smoothed_max_sn",
        "months_expected_min_to_min",
    ):
        require(complete[col].notna().all(),
                f"Missing required completed-cycle label: {col}")

    require(
        not ((cycles["cycle_number"] == 25)
             & (cycles["status"] == "completed")).any(),
        "Cycle 25 must not be treated as a completed cycle.",
    )

    print("\nCycle-wise fold eligibility")
    for held_out in range(13, 25):
        training_cycles = list(range(1, held_out))
        require(max(training_cycles) < held_out,
                f"Cycle leakage at holdout {held_out}.")
        print(
            f"holdout={held_out:2d}; training="
            f"{training_cycles[0]}-{training_cycles[-1]}; "
            f"training_count={len(training_cycles)}"
        )

    print("\nPASS: validation checks completed.")
    print("No models trained. No forecast or report files written.")

#!/usr/bin/env python
"""Run leakage-safe chronological baseline forecasts for monthly SILSO SSN.

Run from any directory:
    python scripts/run_backtesting.py
    python scripts/run_backtesting.py --exclude-provisional-targets

Only the prepared monthly SILSO series is read. Forecasts are evaluated for
target months from January 2016 through the latest prepared observation.
"""
from __future__ import annotations

import argparse
import hashlib
import sys
from pathlib import Path

import matplotlib

matplotlib.use("Agg")

import matplotlib.pyplot as plt
import numpy as np
import pandas as pd

ROOT = Path(__file__).resolve().parent.parent
INPUT_PATH = Path("data") / "processed" / "sunspot_monthly_clean.csv"

def load_monthly(root: Path) -> pd.DataFrame:
    data = pd.read_csv(
        root / INPUT_PATH,
        usecols=["month_start", "ssn", "is_provisional"],
        parse_dates=["month_start"],
    )

    if data["is_provisional"].dtype != bool:

def validate_input(data: pd.DataFrame) -> list[tuple[str, str, str]]:
    checks: list[tuple[str, str, str]] = []
    dates = data["month_start"]
    valid_dates = dates.notna().all() and (dates.dt.day == 1).all()
    checks.append((
        "PASS" if valid_dates else "FAIL",
        "Valid month-start dates",
        f"{len(data)} nonmissing dates, all on day 1" if valid_dates else "Missing or non-month-start dates found",
    ))
    month_index = dates.dt.year * 12 + dates.dt.month
    steps = month_index.diff().dropna()
    continuity = dates.is_monotonic_increasing and not dates.duplicated().any() and bool((steps == 1).all())
    checks.append((

    checks.append((
        "PASS" if provisional_valid else "FAIL",
        "Provisional flags present",
        "every row has a provisional status" if provisional_valid else "missing provisional status found",

    return checks


def build_predictions(
    data: pd.DataFrame,
    horizons: tuple[int, ...] = HORIZONS,
    evaluation_start: pd.Timestamp = EVALUATION_START,
    exclude_provisional_targets: bool = False,
) -> pd.DataFrame:
    """Build target-aligned forecasts at fixed monthly origins.

    Seasonal naive uses the observed value 12 months before the target. Since
    requested horizons are at most 12 months, this source is no later than
    the forecast origin. At a 12-month horizon this definition equals
    persistence; this is expected for a period-12 seasonal-naive forecast.
    Persistence repeats the value at the origin.
    """
    ordered = data.sort_values("month_start").reset_index(drop=True)
    dates = ordered["month_start"]
    values = ordered["ssn"].to_numpy(dtype=float)
    provisional = ordered["is_provisional"].to_numpy(dtype=bool)
    start_idx = int(np.searchsorted(dates.to_numpy(), evaluation_start.to_datetime64()))
    rows = []
    for target_idx in range(start_idx, len(ordered)):
        for horizon in horizons:
            if horizon < 1 or horizon > 12:
                raise ValueError("Supported forecast horizons must be between 1 and 12 months")
            origin_idx = target_idx - horizon
            seasonal_source_idx = target_idx - 12
            if origin_idx < 0 or seasonal_source_idx < 0:
                continue
            origin_month = dates.iloc[origin_idx]
            target_month = dates.iloc[target_idx]
            actual = float(values[target_idx])
            is_provisional = bool(provisional[target_idx])
            included = not (exclude_provisional_targets and is_provisional)
            forecasts = (
                ("persistence", origin_idx, float(values[origin_idx])),
                ("seasonal_naive", seasonal_source_idx, float(values[seasonal_source_idx])),
            )
            for model, source_idx, prediction in forecasts:
                source_month = dates.iloc[source_idx]
                rows.append({
                    "forecast_origin": origin_month,
                    "target_month": target_month,
                    "actual_ssn": actual,
                    "prediction_ssn": prediction,
                    "horizon_months": horizon,
                    "model": model,
                    "is_provisional_target": is_provisional,
                    "included_in_scoring": included,
                    "scoring_exclusion_reason": "provisional_target" if not included else "",
                    "predictor_source_month": source_month,
                    "predictor_source_ssn": float(values[source_idx]),
                })
    return pd.DataFrame(rows).sort_values(
        ["forecast_origin", "horizon_months", "model"]
    ).reset_index(drop=True)


def build_metrics(predictions: pd.DataFrame) -> pd.DataFrame:
    rows = []
    for (model, horizon), group in predictions.groupby(["model", "horizon_months"], sort=True):
        scored = group[group["included_in_scoring"]]
        errors = scored["prediction_ssn"] - scored["actual_ssn"]
        rows.append({
            "model": model,
            "horizon_months": int(horizon),
            "evaluation_start": EVALUATION_START.strftime("%Y-%m"),
            "evaluation_end": group["target_month"].max().strftime("%Y-%m"),
            "scored_observations": len(scored),
            "provisional_targets_retained": int(
                (scored["is_provisional_target"] & scored["included_in_scoring"]).sum()
            ),
            "provisional_targets_excluded": int(
                (group["is_provisional_target"] & ~group["included_in_scoring"]).sum()
            ),
            "mae": float(errors.abs().mean()) if len(errors) else np.nan,
            "rmse": float(np.sqrt(np.square(errors).mean())) if len(errors) else np.nan,
        })
    return pd.DataFrame(rows).sort_values(["horizon_months", "model"]).reset_index(drop=True)


def validate_predictions(predictions: pd.DataFrame, data: pd.DataFrame) -> list[tuple[str, str, str]]:
    checks: list[tuple[str, str, str]] = []
    chronology = bool((predictions["forecast_origin"] < predictions["target_month"]).all())
    checks.append((
        "PASS" if chronology else "FAIL",
        "Forecast chronology",
        "every origin precedes its target" if chronology else "a forecast origin is not before its target",
    ))
    origin_order = predictions["forecast_origin"].is_monotonic_increasing
    checks.append((
        "PASS" if origin_order else "FAIL",
        "Chronological forecast-origin order",
        "prediction rows are ordered by forecast origin" if origin_order else
        "forecast origins are not in chronological order",
    ))

    horizon_alignment = (
        predictions["target_month"].dt.year * 12 + predictions["target_month"].dt.month
        - predictions["forecast_origin"].dt.year * 12 - predictions["forecast_origin"].dt.month
    )
    aligned = bool((horizon_alignment == predictions["horizon_months"]).all())
    checks.append((
        "PASS" if aligned else "FAIL",
        "Target-date alignment",
        "target month is exactly horizon months after origin" if aligned else "origin/target/horizon mismatch",
    ))

    data_by_month = data.set_index("month_start")
    source_available = bool((predictions["predictor_source_month"] <= predictions["forecast_origin"]).all())
    source_dates_exist = bool(predictions["predictor_source_month"].isin(data_by_month.index).all())
    source_values_match = source_dates_exist and all(
        np.isclose(float(row.predictor_source_ssn),
                   float(data_by_month.loc[row.predictor_source_month, "ssn"]), atol=1e-12)
        for row in predictions.itertuples(index=False)
    )
    prediction_matches_source = bool(np.isclose(
        predictions["prediction_ssn"], predictions["predictor_source_ssn"], atol=1e-12
    ).all())
    model_sources_correct = True
    for row in predictions.itertuples(index=False):
        expected_source = row.forecast_origin if row.model == "persistence" else (
            row.target_month - pd.DateOffset(months=12)
        )
        if row.predictor_source_month != expected_source:
            model_sources_correct = False
            break
    leakage_free = (
        source_available and source_dates_exist and source_values_match and
        prediction_matches_source and model_sources_correct
    )
    detail = (
        "all source months and values are valid, source months are available by origin, "
        "and baseline source rules match"
        if leakage_free else
        "future, missing, mismatched, or baseline-inconsistent predictor source found"
    )
    checks.append(("PASS" if leakage_free else "FAIL", "Forecast-origin leakage", detail))

    target_dates_exist = bool(predictions["target_month"].isin(data_by_month.index).all())
    actuals_match = target_dates_exist and all(
        np.isclose(float(row.actual_ssn), float(data_by_month.loc[row.target_month, "ssn"]), atol=1e-12)
        for row in predictions.itertuples(index=False)
    )
    flags_match = target_dates_exist and all(
        bool(row.is_provisional_target) == bool(data_by_month.loc[row.target_month, "is_provisional"])
        for row in predictions.itertuples(index=False)
    )
    checks.append((
        "PASS" if actuals_match and flags_match else "FAIL",
        "Target values and provisional flags",
        "actuals and provisional flags match the prepared data" if actuals_match and flags_match else
        "a target value or provisional flag differs from the prepared data",
    ))
    target_window = bool((predictions["target_month"] >= EVALUATION_START).all())
    checks.append((
        "PASS" if target_window else "FAIL",
        "Locked evaluation window",
        f"all scored target rows start no earlier than {EVALUATION_START:%Y-%m}" if target_window else
        f"target before locked start {EVALUATION_START:%Y-%m} found",
    ))
    paired = predictions.groupby(["target_month", "horizon_months"])["model"].agg(set)
    horizons_covered = set(predictions["horizon_months"].unique()) == set(HORIZONS)
    paired_comparison = horizons_covered and bool((paired.map(frozenset) == frozenset(MODELS)).all())
    checks.append((
        "PASS" if paired_comparison else "FAIL",
        "Fair paired baseline coverage",
        "both baselines cover the same targets at every horizon" if paired_comparison else
        "baseline target coverage differs",
    ))
    seasonal_12 = predictions[
        (predictions["horizon_months"] == 12) &
        (predictions["model"] == "seasonal_naive")
    ]
    persistence_12 = predictions[
        (predictions["horizon_months"] == 12) &
        (predictions["model"] == "persistence")
    ]
    seasonal_12_by_target = seasonal_12.set_index("target_month")["prediction_ssn"]
    persistence_12_by_target = persistence_12.set_index("target_month")["prediction_ssn"]
    equivalent_12 = (
        len(seasonal_12) > 0 and
        seasonal_12_by_target.index.equals(persistence_12_by_target.index) and
        bool(np.isclose(seasonal_12_by_target, persistence_12_by_target, atol=1e-12).all())
    )
    checks.append((
        "PASS" if equivalent_12 else "FAIL",
        "12-month seasonal-naive equivalence",
        "period-12 seasonal naive equals persistence at a 12-month horizon by definition"
        if equivalent_12 else "12-month seasonal-naive predictions differ from persistence",
    ))
    return checks

def markdown_table(frame: pd.DataFrame, floatfmt: str = ".3f") -> str:

def load_monthly_until(path: Path, inclusive_end: pd.Timestamp) -> pd.DataFrame:
    """Read model fields through the cutoff while checking dates through EOF.

    Post-cutoff SSN and provisional values are not parsed or retained.
    """
    rows: list[dict[str, Any]] = []
    previous_month: pd.Timestamp | None = None
    with path.open("r", encoding="utf-8", newline="") as stream:
        reader = csv.DictReader(stream)
        required = {"month_start", "ssn", "is_provisional"}
        if not reader.fieldnames or not required.issubset(reader.fieldnames):
            raise ValueError(f"Prepared monthly input must contain columns: {sorted(required)}")
        for row_number, row in enumerate(reader, start=2):
            month = pd.Timestamp(row["month_start"])
            if pd.isna(month) or month.day != 1:
                raise ValueError(
                    f"Invalid month_start on CSV row {row_number}: {row['month_start']!r}"
                )
            if previous_month is not None and month <= previous_month:
                raise ValueError(
                    "Input dates must be strictly increasing and unique across the CSV; "
                    f"duplicate or out-of-order month on row {row_number}: {month:%Y-%m}"
                )
            previous_month = month
            if month > inclusive_end:
                continue
            flag = row.get("is_provisional")
            normalized_flag = flag.strip().lower() if flag is not None else ""
            if normalized_flag in TRUE_PROVISIONAL_VALUES:
                is_provisional = True
            elif normalized_flag in FALSE_PROVISIONAL_VALUES:
                is_provisional = False
            else:
                raise ValueError(
                    f"Invalid is_provisional value on CSV row {row_number} "
                    f"({month:%Y-%m}): {flag!r}; expected true/false, yes/no, or 1/0"
                )
            rows.append({
                "month_start": month,
                "ssn": float(row["ssn"]),
                "is_provisional": is_provisional,
            })
    result = pd.DataFrame(rows, columns=["month_start", "ssn", "is_provisional"])
    validate_monthly(result, inclusive_end)
    return result

def validate_monthly(data: pd.DataFrame, inclusive_end: pd.Timestamp) -> None:
    if data.empty:
        raise ValueError("No monthly observations were read")
    if data["month_start"].isna().any() or (data["month_start"].dt.day != 1).any():
        raise ValueError("Input dates must be valid first-of-month dates")
    month_numbers = data["month_start"].dt.year * 12 + data["month_start"].dt.month
    if not data["month_start"].is_monotonic_increasing or data["month_start"].duplicated().any():
        raise ValueError("Input dates must be strictly increasing and unique")
    if not bool((month_numbers.diff().dropna() == 1).all()):
        raise ValueError("Input dates must be consecutive monthly observations")
    if data["ssn"].isna().any() or not np.isfinite(data["ssn"].to_numpy(dtype=float)).all():
        raise ValueError("SSN values must be finite and nonmissing")
    if (data["ssn"] < 0).any():
        raise ValueError("SSN values must be nonnegative")
    if data["is_provisional"].isna().any():
        raise ValueError("Provisional flags must be present")
    if data["month_start"].max() > inclusive_end:
        raise ValueError("Input contains observations after the authorized stage cutoff")


def quarter_origins(start: pd.Timestamp, end: pd.Timestamp) -> list[pd.Timestamp]:
    """Return Mar/Jun/Sep/Dec origin months on a fixed quarterly cadence."""
    current = pd.Timestamp(year=start.year, month=1, day=1)
    result = []
    while current <= end:
        if current.month in (3, 6, 9, 12):
            result.append(current)
        current += pd.DateOffset(months=1)
    return result


def stage_targets(
    data: pd.DataFrame,
    stage_start: pd.Timestamp,
    stage_end: pd.Timestamp,
) -> pd.DataFrame:
    """Create target rows from the frozen quarter-end origin schedule."""

    date_to_index = pd.Series(data.index.to_numpy(), index=data["month_start"])
    rows = []
    for origin in origins:
        for horizon in HORIZONS:
            target_month = origin + pd.DateOffset(months=horizon)
            if not stage_start <= target_month <= stage_end:
                continue
            if origin not in date_to_index.index:
                raise ValueError(
                    f"Missing required forecast-origin month: {origin:%Y-%m}"

            origin_idx = int(date_to_index.loc[origin])
            if target_month not in date_to_index.index:
                raise ValueError(f"Missing target month on required schedule: {target_month:%Y-%m}")
            target_idx = int(date_to_index.loc[target_month])
            rows.append({
                "forecast_origin": origin,
                "target_month": target_month,
                "horizon_months": horizon,
                "origin_index": origin_idx,

                "actual_ssn": float(data.loc[target_idx, "ssn"]),

    if not bool((result["forecast_origin"] < result["target_month"]).all()):
        raise AssertionError("Internal schedule error: origin must precede target")

def _roots_pass(roots: Any) -> bool:
    values = np.asarray(roots, dtype=complex)

def classify_warnings(
    captured: list[warnings.WarningMessage],
    phase: str,
) -> tuple[str, str]:
    """Classify warnings by operation; interval warnings never invalidate points."""
    details = []
    fatal_categories = []
    for item in captured:
        category = item.category.__name__
        classification = "nonfatal_diagnostic_warning"
        if category == "ConvergenceWarning":
            classification = (
                "fatal_optimizer_warning" if phase in {"fit", "point_forecast"}
                else "interval_warning_not_point_fatal"
            )
        details.append({
            "phase": phase,
            "category": category,
            "classification": classification,
            "message": str(item.message),
        })
        if classification == "fatal_optimizer_warning":
            fatal_categories.append(category)
    return json.dumps(details, sort_keys=True), ", ".join(sorted(set(fatal_categories)))
